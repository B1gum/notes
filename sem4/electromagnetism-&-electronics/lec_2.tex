\subsection{Kirchoff's Circuit Laws}

\subsubsection{Kirchoff's Current Law (KCL)}
\begin{equation*}
  \sum_{n = 1}^{N} I_n = 0
.\end{equation*}

\begin{symbols}
  \sym{$I_n$}{Current at node $n$}{\unit{\A}}
  \sym{$N$}{Amount of nodes}{---}
\end{symbols}

\subsubsection{Kirchoff's Voltage Law (KVL)}
\begin{equation*}
  \sum_{n = 1}^{N} V_n = 0
.\end{equation*}

\begin{symbols}
  \sym{$V_n$}{Voltage difference between two points}{\unit{\V}}
  \sym{$N$}{Amount of points}{\unit{\N}}
\end{symbols}

\subsubsection{Resistors in Series}
\begin{equation*}
  R_{\mathrm{EQ}} = \sum_{n = 1}^{N} R_n
.\end{equation*}

\begin{symbols}
\InsertSym{Req}
\sym{$R_n$}{Resistance of resistor $n$}{\unit{\ohm}}
\sym{$N$}{Amount of resistors in series}{---}
\end{symbols}

\subsubsection{Voltage Divider Rule}
\begin{equation*}
  V_n = \frac{R_n}{\sum_{n = 1}^{N} R_n} V_s
.\end{equation*}

\begin{symbols}
  \sym{$V_n$}{Voltage difference over resistor $n$}{\unit{\V}}
  \sym{$R_n$}{Resistance over resistor $n$}{\unit{\ohm}}
  \InsertSym{Vs}
\end{symbols}

\subsubsection{Resistors in Parallel}
\begin{equation*}
  R_{\mathrm{EQ}} = \frac{1}{\sum_{n = 1}^{N} \frac{1}{R_n}}
.\end{equation*}

\begin{symbols}
\InsertSym{Req}
\sym{$R_n$}{Resistance of resistor $n$}{\unit{\ohm}}
\sym{$N$}{Amount of resistors in parallel}{---}
\end{symbols}

\subsubsection{Current Divider Rule}
\begin{equation*}
  I_n = \frac{\frac{1}{R_n}}{\sum_{n = 1}^{N} \frac{1}{R_n}} I_s
.\end{equation*}

\begin{symbols}
  \sym{$I_n$}{Current difference over resistor $n$}{\unit{\A}}
  \sym{$R_n$}{Resistance over resistor $n$}{\unit{\ohm}}
  \InsertSym{Is}
\end{symbols}
