\begin{equation*}
  \left\langle x(t) \right\rangle = \frac{1}{T} \int_{0}^{T} x(t') \, \mathrm{d}t'
.\end{equation*}

\subsection{Transient Analysis}

\subsubsection{1st Order Differential Equation for RC Circuits}
\begin{equation*}
  \tau \frac{\mathrm{d}x(t)}{\mathrm{d}t} + x(t) = K_s f(t)
.\end{equation*}

\begin{symbols}
\sym{$\tau$}{$\tau = a_1 / a_0$ is a time constant}{---}
\sym{$x$}{Voltage or current}{---}
\InsertSym{t}
\sym{$K_s$}{$K_s = b_0 / a_0$ is DC gain}{---}
\end{symbols}

\subsubsection{2nd Order Differential Equation for RLC Circuits}
\begin{equation*}
  \frac{1}{\omega_n^2} \frac{\mathrm{d}x^2(t)}{\mathrm{d}t^2} + \frac{2 \zeta}{\omega_n } \frac{\mathrm{d}x(t)}{\mathrm{d}t} = K_s f(t)
.\end{equation*}

\begin{symbols}
\sym{$\omega_n$}{$\omega_n = \sqrt{a_0 / a_2}$ is the natural frequency}{\unit{\Hz}}
\sym{$\zeta$}{$\zeta = a_1 / 2 \sqrt{1 / (a_0 a_2)}$ is the damping ratio}{---}
\sym{$x$}{Voltage or current}{---}
\InsertSym{t}
\sym{$K_s$}{$K_s = b_0 / a_0$ is DC gain}{---}
\end{symbols}
