\subsection{Diodes}

\subsubsection{Net Diode Current in forward bias mode}
\begin{equation*}
  I_D = I_0 \left( e^{V_D \frac{q}{k T}} - 1 \right)
.\end{equation*}

\begin{symbols}
  \sym{$I_D$}{Current across diode}{\unit{\A}}
  \sym{$I_0$}{Leakage current at low voltage}{\unit{\A}}
  \sym{$q$}{Charge of an electron}{\qty{1,6e-19}{\C}}
  \sym{$V_D$}{Voltage across diode}{\unit{\V}}
  \sym{$k$}{Boltzman's constant}{\qty{1,3e-23}{\J\per\K}}
  \sym{$T$}{Temperature}{\unit{\K}}
\end{symbols}
