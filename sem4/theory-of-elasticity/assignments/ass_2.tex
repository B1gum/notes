  \begin{probdef}
    Let us consider a continuum and a Cartesian reference system $(O; \Vec{e}_1, \Vec{e}_2, \Vec{e}_3)$.
    The following displacement vector field is measured:
    \begin{equation}
      \begin{aligned}
        u_1(x_1, x_2, x_3) &= x_1x_3 + x_2^2 + 1, \\
        u_2(x_1, x_2, x_3) &= x_1 x_3^2 + x_1 x_2, \\
        u_3(x_1, x_2, x_3) &= x_1^2 + 1,
      \end{aligned}
    \end{equation}
    where $(x_1, x_2, x_3)$ are the Cartesian coordinates of a point within the continuum.
  \end{probdef}

  \begin{problem}{1}
    Consider three material points on the continuum
    \begin{itemize}
      \item $\Vec{P}_1$, with coordinates: $\begin{pmatrix}
        1 & 0 & 1\\
      \end{pmatrix}$;
      \item $\Vec{P}_2$, with coordinates: $\begin{pmatrix}
        1 & 1 & 1\\
      \end{pmatrix}$;
      \item $\Vec{P}_3$, with coordinates: $\begin{pmatrix}
        0 & 2 & 1\\
      \end{pmatrix}$;
    \end{itemize}
    Then:
    \begin{itemize}
      \item determine the components of the infinitesimal strain and rotation tensor for all points;
      \item determine the principal deformations and principal direction of deformations for all points.
    \end{itemize}
  \end{problem}

  \begin{derivation}
    We start by determining the displacement gradient by finding the first derivatives of the displacement field:
    \begin{align*}
      u_{1,1} &= x_3, & u_{1,2} &= 2x_2, & u_{1,3} &= x_1, \\
      u_{2,1} &= x_3^2 + x_2, & u_{2,2} &= x_1, & u_{2,3} &= 2x_1 x_3, \\
      u_{3,1} &= 2x_1, & u_{3,2} &= 0, & u_{3,3} &= 0
    .\end{align*}
    Meaning
    \begin{equation*}
      \nabla \Vec{u} = [u_{i,j}] = \begin{bmatrix}
        x_3 & 2x_2 & x_1\\
      x_3^2 + x_2 & x_1 & 2x_1 x_3\\
      2x_1 & 0 & 0\\
      \end{bmatrix}
    .\end{equation*}
    The infinitesimal strain tensor $\Vec{\epsilon}$ is given by
    \begin{equation*}
      \epsilon_{ij} = \frac{1}{2} \left( u_{i,j} + u_{j,i} \right) \implies \Vec{\epsilon} \frac{1}{2} \left( \nabla  \Vec{u} + \left( \nabla \Vec{u} \right)^{\intercal} \right)
    \end{equation*}
    and the rotation tensor $\Vec{\omega}$ is given by
    \begin{equation*}
      \omega_{ij}  = \frac{1}{2} \left( u_{i,j} - u_{j,i} \right) \implies \Vec{\omega} = \frac{1}{2} \left( \nabla  \Vec{u} - \left( \nabla \Vec{u} \right)^{\intercal} \right)
    .\end{equation*}
    So
    \begin{align*}
      \Vec{\epsilon} &= \begin{bmatrix}
        x_3 & \frac{x_3^2 + 3x_2}{2} & \frac{3x_1}{2}\\
      \frac{x_3^2 + 3x_2}{2} & x_1 & x_1x_3\\
      \frac{3x_1}{2} & x_1x_3 & 0\\
      \end{bmatrix} \\
      \Vec{\omega} &= \begin{bmatrix}
        0 & \frac{x_2 - x_3^2}{2} & - \frac{x_1}{2}\\
      \frac{x_3^2 - x_2}{2} & 0 & x_1x_3\\
      \frac{x_1}{2} & -x_1x_3 & 0\\
      \end{bmatrix}
    .\end{align*}
    Evaluating these at all three material points yields:

      \frac{1}{2} & 1 & 1\\

      0 & - \frac{1}{2} & - \frac{1}{2}\\
    \frac{1}{2} & 0 & 1 \\
    \frac{1}{2} & -1 & 0\\
    \end{bmatrix}
    .\end{align*}
    The principal strains at $P_1$ is simply the eigenvalues of $\Vec{\epsilon}(P_1)$ and the principal directions are their corresponding eigenvectors.

  \begin{problem}{2}
    Determine the mean normal strain and the strain deviator tensors, for the points $\Vec{p}_1, \Vec{p}_{2}$, and $\Vec{p}_3$.
  \end{problem}

  \begin{derivation}
    The strain $\Vec{\epsilon}$ can be decomposed to a mean normal strain $\Vec{\epsilon}_M$ and a strain deviator tensor $\Vec{\epsilon}_D$ as
    \begin{equation*}
      \Vec{\epsilon} = \Vec{\epsilon}_M + \Vec{\epsilon}_D 
    .\end{equation*}
    The mean normal strain $\Vec{\epsilon}_M$ is defined as
    \begin{equation*}
      \Vec{\epsilon}_M = \frac{1}{3} \left( \epsilon_{11} + \epsilon_{22} + \epsilon_{33} \right) = \frac{1}{3} (x_3 + x_1 + 0) = \frac{x_1 + x_3}{3}
    .\end{equation*}
    And the deviator is
    \begin{equation*}
      \Vec{\epsilon}_D = \Vec{\epsilon} - \Vec{\epsilon}_M \Vec{I} = \begin{bmatrix}
        x_3 - \frac{x_1 + x_3}{3}  & \frac{x_3^2 + 3x_2}{2} & \frac{3x_1}{2}\\
      \frac{x_3^2 + 3x_2}{2} & x_1 - \frac{x_1 + x_3}{3} & x_1 x_3\\
      \frac{3x_1}{2} & x_1x_3 & - \frac{x_1 + x_3}{3}\\
      \end{bmatrix}
    .\end{equation*}
    At $P_1$ we get
    \begin{gather*}
      \Vec{\epsilon}(P_1) = \begin{bmatrix}
        1 & \frac{1}{2} & \frac{3}{2}\\
      \frac{1}{2} & 1 & 1\\
      \frac{3}{2} & 1 & 0\\
      \end{bmatrix} \\
        \Vec{\epsilon}_M(P_1) = \frac{1}{3}(1+1+0) = \frac{2}{3} \\
        \Vec{\epsilon}_D (P_1) = \Vec{\epsilon}(P_1) - \frac{2}{3} \Vec{I} = \begin{bmatrix}
          \frac{1}{3} & \frac{1}{2} & \frac{3}{2}\\
        \frac{1}{2} & \frac{1}{3} & 1\\
        \frac{3}{2} & 1 & - \frac{2}{3}\\
        \end{bmatrix}
    .\end{gather*}
    At $P_2$ we get
    \begin{align*}
      \Vec{\epsilon}(P_2) &= \begin{bmatrix}
        1 & 2 & \frac{3}{2}\\
      2 & 1 & 1\\
      \frac{3}{2} & 1 & 0\\
      \end{bmatrix} \\
      \Vec{\epsilon}_M(P_2) &= \frac{1}{3}(1+1+0) = \frac{2}{3}\\
      \Vec{\epsilon}_D (P_2) &= \Vec{\epsilon}(P_2) - \frac{2}{3}\Vec{I} = \begin{bmatrix}
        \frac{1}{3} & 2 & \frac{3}{2}\\
      2 & \frac{1}{3} & 1\\
      \frac{3}{2} & 1 & -\frac{2}{3}\\
      \end{bmatrix} 
    .\end{align*}
    At $P_3$ we get
    \begin{align*}
      \Vec{\epsilon}(P_3) &= \begin{bmatrix}
        1 & \frac{7}{2} & 0\\
      \frac{7}{2} & 0 & 0\\
      0 & 0 & 0\\
      \end{bmatrix} \\
        \Vec{\epsilon}_M(P_3) &= \frac{1}{3} (1) = \frac{1}{3} \\
        \Vec{\epsilon}_D(P_3) &= \Vec{\epsilon}(P_3) - \frac{1}{3}\Vec{I} = \begin{bmatrix}
          \frac{2}{3} & \frac{7}{2} & 0\\
        \frac{7}{2} & - \frac{1}{3} & 0\\
        0 & 0 & - \frac{1}{3}\\
        \end{bmatrix}
    .\end{align*}
  \end{derivation}

  \begin{problem}{3}
    Discuss whether the small gradient approximation is satisfied for the given displacement field.
    Can the infinitesimal deformation strain be used?
    Hint: Consider evaluating the discarded nonlinear (quadratic) term leading to the infinitesimal strain tensor.
  \end{problem}

  \begin{derivation}
    The Lagrangian strain is
    \begin{equation*}
      \epsilon_{ij}^{L} = \frac{1}{2} \left( u_{i,j} + u_{j,i} + u_{k,i}u_{k,j} \right)
    \end{equation*}
    and the infinitesimal strain is
    \begin{equation*}
      \epsilon_{ij} = \frac{1}{2} \left( u_{i,j} + u_{j,i} \right)
    \end{equation*}
    which means the discarded non-linear term is
    \begin{equation*}
      \Delta = \frac{1}{2} u_{k,i} u_{k,j} = \frac{1}{2} \left( \nabla \Vec{u} \right)^{\intercal} \nabla \Vec{u}
    .\end{equation*}
    Evaluating this at $P_1$ we obtain
    \begin{equation*}
      \Delta (P_1) = \begin{bmatrix}
        3 & \frac{1}{2} & \frac{3}{2}\\
      \frac{1}{2} & \frac{1}{2} & 1\\
      \frac{3}{2} & 1 & \frac{5}{2}\\
      \end{bmatrix}
    .\end{equation*}
    Comparing this with the infinitesimal strain at $P_1$
    \begin{equation*}
      \Vec{\epsilon}(P_1) = \begin{bmatrix}
        1 & \frac{1}{2} & \frac{3}{2}\\
      \frac{1}{2} & 1 & 1\\
      \frac{3}{2} & 1 & 0\\
      \end{bmatrix}
    \end{equation*}
    we see that the discarded quadratic term $\Delta(P_1)$ is not small compared to the infinitesimal strain $\Vec{\epsilon}(P_1)$ meaning the small gradient approximation is \textit{not} satisfied at $P_1$.

    At $P_2$ we get
    \begin{equation*}
      \Delta (P_2) = \begin{bmatrix}
        \frac{9}{2} & 2 & \frac{5}{2}\\
      2 & \frac{5}{2} & 2\\
      \frac{5}{2} & 2 & \frac{5}{2}\\
      \end{bmatrix}
    \end{equation*}
    compared to the infinitesimal strain at $P_2$
    \begin{equation*}
      \Vec{\epsilon}(P_2) = \begin{bmatrix}
        1 & 2 & \frac{3}{2}\\
      2 & 1 & 1\\
      \frac{3}{2} & 1 & 0\\
      \end{bmatrix}
    \end{equation*}
    the discarded quadratic term is not small compared to the infinitesimal strain meaning the small gradient approximation is, once again, \textit{not} applicable at $P_2$.

    At $P_3$ we get
    \begin{equation*}
      \Delta (P_3) = \begin{bmatrix}
        5 & 2 & 0\\
      2 & 8 & 0\\
      0 & 0 & 0\\
      \end{bmatrix}
    \end{equation*}
    which is several times larger than the infinitesimal strain at $P_3$
    \begin{equation*}
      \Vec{\epsilon}(P_3) = \begin{bmatrix}
        1 & \frac{7}{2} & 0\\
      \frac{7}{2} & 0 & 0\\
      0 & 0 & 0\\
      \end{bmatrix}
    \end{equation*}
    meaning the small gradient approximation is \textit{not} satisfied at $P_3$ either.
  \end{derivation}

  \finalresult{\text{The small gradient approximation is not satisfied for }P_1, P_2 \text{ or } P_3}
