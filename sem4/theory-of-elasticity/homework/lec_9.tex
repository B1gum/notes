\begin{problem}{4.9}
  Consider an isotropic elastic medium subject to a hydrostatic state of stress:
  \begin{equation*}
    \sigma^{(1)} = \sigma^{(2)} = \sigma^{(3)} = -p
  .\end{equation*}
  Show that for this state of stress
  \begin{equation*}
    p = -KR_1
  ,\end{equation*}
  where $K$ is the bulk modulus and $R_1$ is the cubical strain.
  Consider the implications of $\nu > 1 / 2$ and $\nu < -1$.
\end{problem}

\begin{problem}{4.10}
  For an elastic medium subject to a state of stress $\sigma_{ij}$, assume that the deformation is incompressible for $\sigma_{(ii)} \neq 0$.
  \begin{enumerate}[label=(\alph*)]
    \item Verify that $\nu = \num{0,5}$.
    \item Assume also that $\epsilon_{33} = 0$ and determine $\nu$.
  \end{enumerate}
\end{problem}

\begin{problem}{5.1a}
  The stress components at a point $P(x,y,z)$ in an isotropic medium are
  \begin{align*}
    \sigma_{x x} &= y + 3z^2, & \sigma_{xy} &= z^3, \\
    \sigma_{yy} &= x + 2z, & \sigma_{xz} &= y^2, \\
    \sigma_{zz} &= 2x + y, & \sigma_{yz} = x^2
  .\end{align*}
  No body forces are present.

  Does this stress field satisfy equilibrium?
\end{problem}

\begin{problem}{5.3}
  For the following stress field,
  \begin{align*}
    \sigma_{x x} &= 80x^3 + y, & \sigma_{xy} &= \num{1000} + \num{100} y^2 , \\
    \sigma_{yy} &= \num{100} x^3 + \num{1600} , & \sigma_{xz} &= 0, \\
    \sigma_{zz} &= \num{90} y^2 + \num{100} z^3, & \sigma_{yz} = xz^3 + \num{100} x^2 y
  .\end{align*}

  \begin{enumerate}
    \item[(a)] What is the body force distribution required for equilibrium?
    \item[(c)] With $E = \num{30e6} (\text{force}/\text{length}^2)$ and $\nu = \num{0,3}$, what is the strain at $(2,2,5)$?
  \end{enumerate}
\end{problem}
