\lecture{5}{23. Februar 2026}{The Strain Tensor}

\begin{problem}{3.1}
  The components of a displacement field are
  \begin{align*}
    u_x &= \left( x^2 + 20 \right)\cdot \num{e-4} , \\
    u_y &= 2yz \cdot \num{e-3}  ,\\
    u_z &= \left( z^2 - xy \right) \cdot \num{e-3} 
  .\end{align*}

  \begin{enumerate}[label=(\alph*)]
    \item Consider two points in the undeformed system, $(2,5,7)$ and $(3,8,9)$.
      Find the change in distance between these points.
  \end{enumerate}
\end{problem}

\begin{derivation}
  The displacement at $A = (2,5,7)$ is
  \begin{align*}
    u_x(A) &= \left( 2^2 + 20 \right)\cdot  \num{e-4} = \num{0.0024}  \\
    u_y(A) &= 2 \cdot 5 \cdot 7 \cdot \num{e-3} = \num{0,070}   \\
    u_z(A) &= (7^2 - 2 \cdot 5) \cdot \num{e-3} = \num{0,039}  \\
  \end{align*}
  so
  \begin{equation*}
    \Vec{u}(A) = \begin{pmatrix}
      \num{0,0024} \\
    \num{0,070} \\
    \num{0,039} \\
    \end{pmatrix}
  .\end{equation*}
  And at $B = (3,8,9)$ it is
  \begin{align*}
    u_x(B) &= (3^2 + 20) \cdot \num{e-4} = \num{0,0029}  \\
    u_y(B) &= 2 \cdot 8 \cdot 9 \cdot \num{e-3} = \num{0,144}  \\
    u_z(B) &= (9^2 - 3 \cdot 8) \cdot \num{e-3} = \num{0,057} 
  \end{align*}
  so
  \begin{equation*}
    \Vec{u}(B) = \begin{pmatrix}
      \num{0,0029} \\
    \num{0,144} \\
    \num{0,057} \\
    \end{pmatrix}
  .\end{equation*}
  
  The deformed positions are:
  \begin{align*}
    A' &= A + \Vec{u}(A) = \begin{pmatrix}
      2\\
    5\\
    7\\
    \end{pmatrix} + \begin{pmatrix}
      \num{0,0024} \\
    \num{0,070} \\
    \num{0,039} \\
    \end{pmatrix} = \begin{pmatrix}
      \num{2,0024} \\
    \num{5,070} \\
    \num{7,039} \\
    \end{pmatrix} \\
      B' &= B + \Vec{u}(B) = \begin{pmatrix}
        3\\
      8\\
      9\\
      \end{pmatrix} + \begin{pmatrix}
        \num{0,0029} \\
      \num{0,144} \\
      \num{0,057} \\
      \end{pmatrix} = \begin{pmatrix}
        \num{3,0029} \\
      \num{8,144} \\
      \num{9,057} \\
      \end{pmatrix}
  .\end{align*}
  Our position vectors are:
  \begin{align*}
    AB &= \begin{pmatrix}
      1\\
    3\\
    2\\
    \end{pmatrix} \\
    AB' &= \begin{pmatrix}
     1.0005 \\
     3.074 \\
     2.018 \\
    \end{pmatrix}
  .\end{align*}
  These have the lengths:
  \begin{align*}
    \left| AB \right| &= \sqrt{1^2 + 3^2 + 2^2}  = 3.74166\\
    \left| AB' \right| &=  \sqrt{\num{1,0005}^2  + \num{3,074}^2  + \num{2,018}^2 }  = 3.81088
  .\end{align*}
  The change in length is then:
  \begin{equation*}
    \Delta L = \left| AB' \right|- \left| AB \right| = \num{3,81088} - \num{3,74166}  = \num{0.06922}
  .\end{equation*}
\end{derivation}

\finalresult{\Delta L = \num{0,06922}}
