  \begin{problem}{1}
    Solve the following boundary value problem using the direct method.
    \begin{equation} \label{eq:prob1}
      \frac{\partial^2 u}{\partial x^2}  + \frac{\partial ^2u }{\partial x \partial y} + \frac{\partial ^2 u }{\partial y^2} = f(x,y)
    \end{equation}
    where
    \begin{equation}
      f(x,y) = e^{- \frac{( x-15 )^2}{4}} + e^{ - \frac{(y-15)^2}{4}}
    .\end{equation}
    
    Boundary conditions at the edges are as follows
    \begin{align*}
      u(0,y) &= 0 & u(L_x, y) &= 20 \\
      \frac{\partial u}{\partial y} \bigg|_{y = 0} &= 0 & \frac{\partial u}{\partial y} \bigg|_{y=L_y} &= 1
    .\end{align*}
    Using central difference scheme for all derivatives try to find the steady state response of the system.
    Discretize both $x$ and $y$ domain by 90 meshes for $L_x = L_y = 30$.

    Note: Use the following to plot the calculated $u$ in 3D:
    \begin{verbatim}
      surface(X,Y,u');
      view(3)
    \end{verbatim}
    \vspace{-1.5em}
  \end{problem}

  \begin{derivation}
    We are told to use 90 meshes in both directions
    \begin{equation}
      L_x = L_y = 30
    \end{equation}
    which corresponds to a step-length of 
    \begin{equation}
      h = \frac{30}{90} = \frac{1}{3}
    \end{equation}  
    and
    \begin{equation}
      N = 90 + 1 = 91
    \end{equation}
    grid points, corresponding to $x_i = ih, i = 0,1,2,\ldots ,89,90$ and $y_j = jh, j = 0,1,2,\ldots ,89,90$.
    As we are using the direct method, we will seek to solve the discrete approximation
    \begin{equation}
      A \Vec{u} = \Vec{b}
    \end{equation}
    where
    \begin{equation}
      \Vec{u} = \begin{bmatrix}
        U_{1,1}\\
      U_{2,1}\\
      U_{3,1}\\
      \vdots\\
      \end{bmatrix}
    \end{equation}
    is a vector containing all unknown grid values of $u$, and $A$ contains all the finite-difference stencil coefficients.
    We have in total $91^2 = \num{8281} $ grid points, meaning the $\Vec{A}$-matrix must be of size $\num{8281} x \num{8281}$.

    At any interior grid point, we can approximate the second derivative in $x$, $y$ and the derivative in $x$ and $y$ as
    \begin{align}
      \frac{\partial ^2 u}{\partial x^2} &\approx \frac{U_{i+1,j} - 2U_{i,j} + U_{i-1,j}}{h^2} \label{eq:xapp}\\
      \frac{\partial ^2 u}{\partial y^2} &\approx \frac{U_{i,j+1} - 2U_{i,j} + U_{i,j-1}}{h^2} \label{eq:yapp}\\
      \frac{\partial ^2 u}{\partial x \partial y} &\approx \frac{U_{i+1,j+1} - U_{i+1,j-1} - U_{i-1,j+1} + U_{i-1,j-1}}{4h^2} \label{eq:xyapp}
    .\end{align}
    Inserting \cref{eq:xapp,eq:yapp,eq:xyapp} into \cref{eq:prob1}, and defining $f_{i,j} = f(x_i, y_i)$ yields:
    \begin{align}
      \frac{U_{i+1,j} - 2U_{i,j} + U_{i-1,j}}{h^2} + \frac{U_{i+1,j+1} - U_{i+1,j-1} - U_{i-1,j+1} + U_{i-1, j-1}}{4h^2} + \frac{U_{i,j+1} - 2U_{i,j} + U_{i,j-1}}{h^2} &= f_{i,j} \nonumber\\
      U_{i+1,j} - 2U_{i,j} + U_{i-1,j} + \frac{1}{4} U_{i+1,j+1} - \frac{1}{4} U_{i+1,j-1} - \frac{1}{4} U_{i-1,j+1} + \frac{1}{4} U_{i-1,j-1} + U_{i,j+1} - 2U_{i,j} + U_{i,j-1} &= h^2 f_{i,j} \nonumber\\
      4U_{i,j} - U_{i+1,j} - U_{i-1, j} - U_{i,j+1} - U_{i,j-1} - \frac{1}{4} U_{i+1,j+1} + \frac{1}{4} U_{i+1,j-1} + \frac{1}{4} U_{i-1, j+1} - \frac{1}{4} U_{i-1,j-1} &= - h^2 f_{i,j}
    .\end{align}
    As a stencil, this looks like
    \begin{equation*}
      \begin{array}{ccc}
        \frac{1}{4} & -1 & - \frac{1}{4}\\
      -1 & 4 & -1\\
      - \frac{1}{4} & -1 & \frac{1}{4}\\
      \end{array}
    .\end{equation*}

    The left boundary condition is a Dirichlet-type
    \begin{equation}
      u(0,y) = 0 \implies U_{0,j} = 0, j = 0,1,2,\ldots ,89,90
    \end{equation}
    and the right boundary condition is also a Dirichlet-type boundary condition
    \begin{equation}
      u(L_x, y) = 20 \implies U_{90,j} = 20, j = 0,1,2,\ldots ,89,90
    .\end{equation}
    
    At the bottom and top we are given Von Neumann boundary conditions. 
    At the bottom,
    \begin{equation*}
      \frac{\partial u}{\partial y} \bigg|_{y = 0} = 0
    \end{equation*}
    now, using a central difference at $j = 0$ we get
    \begin{equation} \label{eq:ghost}
      \frac{U_{i,1} - U_{i,-1}}{2h} = 0
    .\end{equation}
    In \cref{eq:ghost} the point $U_{i,-1}$ is outside of the region, we however quickly see that for \cref{eq:ghost} to hold, $U_{i,1} = U_{i,-1}$.
    That means the bottom boundary condition becomes
    \begin{align}
      \frac{U_{i+1,0} - 2U_{i,0} + U_{i-1,0}}{h^2} + \frac{2U_{i,1} - 2U_{i,0}}{h^2} &= f_{i,0} \nonumber\\
      4U_{i,0} - U_{i+1,0} - U_{i-1,0} - 2U_{i,1} &= h^2 f_{i,0} \quad i = 1, \ldots, 89
    .\end{align}

    Following a similar procedure at the top we get:
    \begin{gather}
      \frac{\partial u}{\partial y} \bigg|_{y = L_x} = 1 \nonumber\\
      \frac{U_{i,91} - U_{i,89}}{2h} = 1 \nonumber\\
      U_{i,91} = U_{i,89} + 2h \nonumber\\
      \frac{U_{i+1,90} - 2U_{i,90} + U_{i-1, 90}}{h^2} + \frac{2U_{i,89} - 2U_{i,90} + 2h}{h^2} = f_{i,90} \nonumber\\
      \begin{aligned}
            4 U_{i,90} - U_{i+1,90} - U_{i-1,90} - 2U_{i,89} &= -h^2 f_{i,90} + 2h \\
                                                     &= -h^2 f_{i,90} + \frac{2}{3}
      .\end{aligned}
    \end{gather}

    Based on this, the Matlab code attached in \verb#NumAna_NoahRBH_2026_prob_1.m# has been written.
    Running this code results in \cref{fig:prob_1_sol}.

      \centering
      \includegraphics[width=0.8\linewidth]{./figures/prob_1_sol.png}
      \caption{Steady-state solution to Problem 1, from Matlab using the direct method.}
      \label{fig:prob_1_sol}
    \end{figure}
  \end{derivation}

  \begin{problem}{2}
    In medical physics, the transport of a drug through a specific segment of an artery is modeled using the Advection-Diffusion-Reaction (ADR) Equation:
    \begin{equation*}
      \frac{\partial u}{\partial t}  + \alpha \frac{\partial u}{\partial x}  = \nu \frac{\partial ^2 u}{\partial x^2} - cu
    .\end{equation*}
    In this model
    \begin{itemize}
      \item $\alpha$ represents the speed of blod flow.
        It is the primary mechanism that physically ``pushes'' the drug particles downstream through the arterial segment.
      \item $\nu$ is the Drug Dispersion Coefficient which accounts for the spreading of drug concentration.
        It combines molecular diffusion with ``Taylor dispersion'', which is the mixing effect caused by the blood's velocity profile as it interacts with the vessel walls.
      \item $c$ is the Drug Concentration Decay Rate representing a linear sink term.
        In pharmacokinetics, this models how the drug is ``lost'' from the system, typically through metabolic breakdown by the liver or absorption into the surrounding vascular tissues.
    \end{itemize}
    Assume the blood stream velocity $\alpha = \qty{1}{\cm\per\s}$, drug dispersion coefficient $\nu = \qty{0,1}{\cm^2\per\s} $, and drug concentration decay rate $c = \qty{0,05}{\s^{-1}}$.
    We want to study the ADR for 20 seconds within \qty{20}{\cm} length of the artery from where the drug is injected considering the following initial and boundary conditions:
    \begin{itemize}
      \item \textbf{Initial Conditions:} at $t = 0$ an initial distribution of drug exists in the artery
        \begin{equation*}
          u(x,0) = e^{-(x-5)^2}
        .\end{equation*}
      \item \textbf{Boundary Conditions:} at $x = 0$ a medical pump provides a variable dosage
        \begin{equation*}
          u(0,t) = \sin^2(t)
        \end{equation*}
        at $x = 20$ a ``free-exit'' condition where the drug leaves the segment without resistance
        \begin{equation*}
          \frac{\partial u}{\partial x} \bigg|_{x = 20} = \num{-0,2} 
        .\end{equation*}
    \end{itemize}
    Using implicit Crank-Nicholson Scheme try to solve the problem.
    Use a time step of $\Delta t = \num{0,01}$ to discretize the time domain and 100 mesh to discretize the space domain (number of grid points in space: $N = 101$).

    Animate the results for 20 seconds and plot $u(x,y)$ at the time $t = 2, t = 5, t = 7, t= 10, t = 12, t = 15, t = 18, $ and $t = \qty{20}{\s}$.
  \end{problem}

  \begin{derivation}
    The segment of the artery which we are modelling is
    \begin{equation}
      0 \leq x \leq 20
    \end{equation}
    for times
    \begin{equation}
      0 \leq t \leq 20
    .\end{equation}
    A mesh of 100 should be employed corresponding to
    \begin{equation} \label{eq:2N}
      N = 100 + 1 = 101
    \end{equation}
    grid points.
    The spatial step-size is therefore:
    \begin{equation} \label{eq:2h}
      h = \Delta x = \frac{20 - 0}{101- 1} = \num{0,2} 
    \end{equation}
    and the time step is
    \begin{equation} \label{eq:2dt}
      \Delta t = \num{0,01}
    ,\end{equation}
    which corresponds to 
    \begin{equation} \label{eq:2nt}
      N_t = \frac{20}{\num{0,01}} = \num{2000} 
    \end{equation}
    time steps.
    Based on \cref{eq:2N,eq:2h}, we define the grid points
    \begin{equation}
      x_i = (i-1) h, \qquad i = 1,2,\ldots ,N
    \end{equation}
    and based on \cref{eq:2nt,eq:2dt}, we define the time points
    \begin{equation}
      t_n = n \Delta t, \qquad n = 0,1,\ldots ,N_t
    .\end{equation}
    We will seek to approximate the solution of $u(x,t)$ bu a discrete table of values
    \begin{equation} \label{eq:2solform}
      u_{i}^{n} = u(x_i, t_n)
    .\end{equation}
    
    
    At $t_0$ our initial condition is,
    \begin{equation}
      u(x,0) = e^{- \left( x-5) \right)^2}
    \end{equation}
    which on the form of \cref{eq:2solform} is
    \begin{equation*}
      u_i^{0} = e^{- \left( x-5 \right)^2}
    .\end{equation*}
    
    The left boundary condition is a Dirichlet boundary condition prescribing
    \begin{equation}
      u(0,t) = \sin^2(t)
    \end{equation}
    on the same form as \cref{eq:2solform} this corresponds to
    \begin{equation*}
      u_1^{n} = \sin^2 (t)
    .\end{equation*}

    At $x = 20$, we have a Von Neumann boundary condition,
    \begin{equation}
      \frac{\partial u}{\partial x} \bigg|_{x = 20} = -\num{0,2} 
    .\end{equation}
    Since $x = 20$ we have to use something other than the central difference if we want to avoid points outside our defined interval (as was the case for Problem 1). 
    Using a Taylor-expansion the second-order backward difference approximation is
    \begin{equation*}
      u_x(x_N) = \frac{3u_N^{n} - 4 u_{N-1}^{n} + u_{N-2}^{n}}{2h}
    .\end{equation*}
    So our Von Neumann boundary condition becomes
    \begin{equation*}
      \frac{3u_N^{n} - 4u_{N-1}^{n} + u_{N-2}^{n}}{2h} = \num{-0,2} \implies 3u_N^{n} - 4u_{n-1}^{n} + u_{N-2}^{n} = \num{-0,08}
    .\end{equation*}
    
    For the interior points, i.e. $i = 2, 3, \ldots , N-1$, we approximate the first derivative by a central difference
    \begin{equation} \label{eq:21ord}
      u_x(x_i, t_n) \approx \frac{u_{i+1}^{n} - u_{i-1}^{n}}{2h}
    \end{equation}
    and the second derivative, we approximate by
    \begin{equation} \label{eq:22ord}
      u_{x x }(x_i, t_n) \approx \frac{u_{i+1}^{n} - 2u_{i}^{n} + u_{i-1}^{n}}{h^2}
    .\end{equation}
    Now, rewriting the given PDE as
    \begin{equation} \label{eq:2pde}
      \frac{\partial u}{\partial t} =- \alpha \frac{\partial u}{\partial x} + \nu \frac{\partial ^2 u}{\partial x^2} - cu
    \end{equation}
    and substituting \cref{eq:21ord,eq:22ord} into \cref{eq:2pde} we get the spatial operator
    \begin{equation*}
      L_h u_i = - \alpha \frac{u_{i+1} - u_{i-1}}{2h} + \nu \frac{u_{i+1} - 2u_i + u_{i-1}}{h^2} - cu
    .\end{equation*}
    So the PDE becomes:
    \begin{equation*}
      \frac{\mathrm{d}u_i}{\mathrm{d}y} = L_h u_i
    .\end{equation*}
    
    Now, we use Crank-Nicholson in time.
    Here, we average the spatial operator between the old and new time level,
    \begin{equation} \label{eq:2crank}
      \frac{u_i^{n+1} - u_i^{n}}{\Delta t} = \frac{1}{2} \left( L_h u_i^{n+1} + L_h u_i^{n} \right)
    .\end{equation}
    Rearranging \cref{eq:2crank}, we get
    \begin{equation*}
      u_i^{n+1}- \frac{\Delta t}{2}L_h u_i^{n+1} = u_i^{n} + \frac{\Delta t}{2} L_h u_i^{n}
    .\end{equation*}
    For interior points, we define
    \begin{align}
      \mu &= \frac{\nu \Delta t}{2h^2} \\
      \lambda &= \frac{\alpha \Delta t}{4h} \\
      \rho &= \frac{c \Delta t}{2}
    \end{align}
    using the values given in the problem we obtain
    \begin{gather*}
      \mu = \frac{\num{0,1} \cdot \num{0,01} }{2 \cdot \left( \num{0,2}  \right)^2} = \num{0,0125} \\
      \lambda = \frac{1 \cdot \num{0,01} }{4 \cdot \num{0,2} } = \num{0,0125} \\
      \rho = \frac{\num{0,05} \cdot \num{0,01} }{2} = \num{0,00025} 
    .\end{gather*}
    The Crank-Nicholson equation for an interior point is
    \begin{equation}
      - \left( \mu + \lambda \right)u_{i-1}^{n+1} + \left( 1 + 2\mu + \rho \right)u_i^{n+1} + \left( \lambda - \mu \right)u_{i+1}^{n+1} = \left( \mu + \lambda \right)u_{i-1}^{n} + \left( 1 - 2\mu - \rho \right)u_i^{n} + \left( \mu - \lambda \right)u_{i+1}^{n}
    .\end{equation}
    As $\mu = \lambda$, $\mu - \lambda = 0$, so the above simplifies to:
    \begin{equation}
      - \left( \mu + \lambda \right)u_{i-1}^{n+1} + \left( 1 + 2\mu + \rho \right)u_i^{n+1} = \left( \mu + \lambda \right)u_{i-1}^{n} + \left( 1 - 2\mu - \rho \right)u_i^{n}
    .\end{equation}
  \end{derivation}
