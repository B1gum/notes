\begin{problemwithimage}{p10_1}{P10-1}
  Determine the resultant internal loadings acting on the cross section at point $E$ of the handle arm, and on the cross section of the short link $BC$.
\end{problemwithimage}

\begin{derivation}
  \begin{figure} [ht]
    \centering
    \includegraphics[width=0.5\linewidth]{./figures/p10_1fbd.png}
    \caption{Free body diagram of the hole punch.}
    \label{fig:p10_1fbd}
  \end{figure}
  We start by determining the force $F_{BC}$ in the link $BC$ using a moment equilibrium about $A$, using \Cref{fig:p10_1fbd} we get
  \begin{align*}
    R_{BC} \cdot \cos \ang{30} \cdot \qty{50}{\mm} &- \qty{120}{\N} \cdot \qty{500}{\mm} = 0 \\
    R_{BC} &= \frac{\qty{120}{\N} \cdot \qty{500e-3}{\m} }{\cos \ang{30} \cdot \qty{50e-3}{\m} } \\
           &= \qty{1,38564}{\kN} 
  .\end{align*}
  
  \begin{figure} [ht]
    \centering
    \includegraphics[width=0.15\linewidth]{./figures/p10_1fbd1.png}
    \caption{Free body diagram of $BC$}
    \label{fig:p10_1fbd1}
  \end{figure}
  From \Cref{fig:p10_1fbd1} we see that
  \[ 
  -R_{BC} + N_{BC} = 0 \implies R_{BC} = N_{BC} = \qty{1,38564}{\kN} 
  .\]
  This link will not experience any shear force, torque or bending moment and the internal loadings are therefore all found.

  \begin{figure} [ht]
    \centering
    \includegraphics[width=0.25\linewidth]{./figures/p10_1fbd2.png}
    \caption{Free body diagram of the segment}
    \label{fig:p10_1fbd2}
  \end{figure}
  From \Cref{fig:p10_1fbd2} we see that
  \[ 
  \sum F_x = 0 \implies N_E = 0
  .\]
  Likewise, in the $y$-direction we get
  \[ 
  \sum F_y = 0 \implies V_E = \qty{120}{\N} 
  .\]
  The moment in $E$ $M_E$ must balance the moment from the applied force
  \[ 
  - M_E - \qty{120}{\N} \cdot \qty{400}{\mm} = 0 \implies M_E = \qty{120}{\N} \cdot \qty{400e-3}{\m} = \qty{48}{\N\m} 
  .\]
\end{derivation}

\finalresult{\begin{aligned}
  N_{BC} &= \qty{1,38564}{\kN} \\
  V_{E} &= \qty{120}{\N}  \\
  M_{E} &= \qty{48}{\N\m}
\end{aligned}}


\begin{problemwithimage}{p10_2}{P10-2}
  The bar has a cross-sectional area of \qty{400e-6}{\square\m}. 
  If it is subjected to a triangular axial distributed loading along its length which is 0 at $x = 0$ and \qty{9}{\kN\per\m} at $x = \qty{1,5}{\m}$, and to two concentrated loads as shown, determine the average normal stress in the bar as a function of $x$ for $0 \leq x \leq \qty{0,6}{\m}$.
\end{problemwithimage}

\begin{figure} [ht]
  \centering
  \includegraphics[width=0.5\linewidth]{./figures/p10_2fbd.png}
  \caption{Free body diagram of the bar.}
\end{figure}
\begin{derivation}
  We consider the right segment of the bar from $x$ to $\qty{1,5}{\m}$.
  A force balance in the $x$-direction with positive to the right yields
  \begin{align*}
    N &= \qty{8}{\kN} + \qty{4}{\kN} + \frac{1}{2} \left( \qty{6}{\kN\per\m} \cdot x + \qty{9}{\kN}  \right) \left( \qty{1,5}{\m} - x \right) \\
     &= \qty{18,75}{\kN} - \qty{3}{\kN\per\square\m} \cdot x^2
  .\end{align*}
  The average normal stress is simply defined as
  \[ 
  \sigma_{\mathrm{avg}} = \frac{N}{A}
  ,\]
  hence,
  \begin{align*}
    \sigma_{\mathrm{avg}}(x) &= \frac{\qty{18,75}{\kN}}{\qty{400e-6}{\m\squared}} - \frac{\qty{3}{\kN\per\square\m}}{\qty{400e-6}{\square\m} } x^2 \\
    &= \qty{46,875}{\MPa} - \qty{7,5}{\MPa\per\square\m} \cdot x^2
  .\end{align*}
\end{derivation}

\finalresult{\sigma_{\mathrm{avg}}(x) = \qty{46,875}{\MPa} - \qty{7,5}{\MPa\per\square\m} \cdot x^2}


\begin{problemwithimage}{p11_1}{P11-1}
  Determine the required thickness of member $BC$ to the nearest \unit{\mm}, and the diameter of the pins at $A$ and $B$ if the allowable normal stress for member $BC$ is $\sigma_{\text{allow}} = \qty{200}{\MPa}$ and the allowable shear stress for the pins is $\tau_{\text{allow}} = \qty{70}{\MPa}$.
\end{problemwithimage}

\begin{figure} [ht]
  \centering
  \includegraphics[width=0.85\linewidth]{./figures/p11_1-1.png}
  \caption{Free body diagram of the structure.}
  \label{fig:p11_1-1}
\end{figure}

\begin{derivation}
  We start by converting the distributed load into an equivalent point load, this load will be
  \[ 
  P = \qty{30}{\kN\per\m} \cdot \qty{2,4}{\m} = \qty{72}{\kN} 
  .\]
  This load will be placed at the middle of the member, \qty{1,2}{\m} from $A$ as shown on \Cref{fig:p11_1-1}.
  We start with the moment equilibrium about point $A$ as
  \begin{align*}
    \sum M_A &= 0 \\
    0 &= P \cdot \qty{1,2}{\m} - F_{BC}\cdot \sin \ang{60} \cdot \qty{2,4}{\m} \\
    F_{BC} &= \frac{\qty{72}{\kN} \cdot \qty{1,2}{\m} }{\sin \ang{60} \cdot \qty{2,4}{\m}} \\
           &= \qty{41,57}{\kN} 
  .\end{align*}
  Now a force equilibrium in the $x$-direction yields
  \begin{align*}
    \sum F_x &= 0 \\
    0 &= F_{BC}\cdot \cos \ang{60} - A_x \\
    A_x &= F_{BC} \cdot \cos \ang{60} \\
        &= \qty{41,57}{\kN} \cdot \cos \ang{60} = \qty{20,78}{\kN} 
  .\end{align*}
  In the $y$-direction we get
  \begin{align*}
    \sum F_y &= 0 \\
    0 &= F_{BC} \cdot \sin \ang{60} + A_y - P \\
    A_y &= F_{BC}\cdot \sin \ang{60} - P \\
      &= \qty{72}{\kN} - \qty{41,57}{\kN} \cdot \sin \ang{60} = \qty{36}{\kN} 
  .\end{align*}
  This corresponds to a total reaction at $A$ of
  \[ 
  R_A = \sqrt{A_x^2 + A_y^2} = \sqrt{\left( \qty{20,78}{\kN}  \right)^2 + \left( \qty{36}{\kN}  \right)^2} = \qty{41,57}{\kN}
  .\]
  The required thickness of $BC$ can be found from the allowable normal stress as
  \begin{align*}
    \sigma_{\text{allow}} &= \frac{F_{BC}}{A_{BC}} \\
    A_{BC} &= \frac{F_{BC}}{\sigma_{\text{allow}}} \\
    &= \frac{\qty{41,57}{\kN}}{\qty{200}{\MPa}} \\
    &= \qty{2,078e-4}{\m\squared} \\
    A_{BC} &= w_{BC} \cdot h_{BC} \\
    w_{BC} &= \frac{A_{BC}}{h_{C}} \\
      &= \frac{\qty{2,078e-4}{\m\squared }}{\qty{0,04}{\m}} \\
      &= \qty{5,2}{\mm}
  .\end{align*}
  A can be seen to be a single-stress shear pin on the figure, hence
  \begin{align*}
    \tau_{\text{allow}} &= \frac{R_A}{A_{A}} \\
    A_{A} &= \frac{R_A}{\tau_{\text{allow}}} \\
    &= \frac{\qty{41,57}{\kN}}{\qty{70}{\MPa} } \\
    &= \qty{5,9e-4}{\square\m} \\
    A_{A} &= \pi \cdot \frac{D_A^2}{4} \\
    D_{A} &= \sqrt{\frac{4 \cdot A_{A}}{\pi}} \\
          &= \sqrt{\frac{4 \cdot \qty{5,9e-4}{\square\m}  }{\pi}} \\
          &= \qty{27}{\mm} 
  .\end{align*}
  And as $B$ is a double-stress shear pin we get
  \begin{align*}
    \tau_{\text{allow}} &= \frac{F_{BC}}{2 A_{B}} \\
    A_B &= \frac{2 \cdot \tau_{\text{allow}}}{F_{BC}} \\
        &= \frac{\qty{41,57}{\kN}}{2 \cdot \qty{70}{\MPa} } \\
    &=   \qty{3e-4}{\m\squared} \\
    A_{B} &= \pi \cdot \frac{D_B^2}{4}  \\
    D_B &= \sqrt{\frac{4 \cdot A_B}{\pi}} \\
    &= \sqrt{\frac{4 \cdot \qty{3e-4}{\m\squared} }{\pi}} \\
    &= \qty{19}{\mm} 
  .\end{align*}
\end{derivation}

\finalresult{
  \begin{aligned}
    w_{BC} &= \qty{5,2}{\mm} \\
    D_A &= \qty{27}{\mm}  \\
    D_B &= \qty{19}{\mm}
  .\end{aligned}
}

\begin{problemwithimage}{p11_2}{P11-2}
  Nylon strips are fused to glass plates. When moderately heated the nylon will become soft while the glass stays approximately rigid. Determine the average shear strain in the nylon due to the load $\Vec{P}$ when the assembly deforms as indicated.
\end{problemwithimage}

\begin{derivation}
  As the glass is approximately rigid we can assume it does not deform at all and that all of the material deformation is due to the nylon deforming.

  \begin{figure} [ht]
    \centering
    \includegraphics[width=0.75\linewidth]{./figures/p11_2fbd.png}
    \caption{Free body diagram of the system.}
    \label{fig:p11_2fbd}
  \end{figure}

  Shear strain is defined as the angle the material deviates from perpendicular.
  This can be thought of as a right triangle giving:
  \begin{align*}
    \tan \gamma &= \frac{\qty{2}{\mm} }{\qty{10}{\mm} } \\
    \gamma &= \tan^{-1} \left( \frac{2}{10} \right) \\
           &= \num{0,197}
  .\end{align*}
\end{derivation}
\finalresult{\gamma = \num{0,197}}


\begin{problemwithimage}{p12_1}{P12-1}
  The rigid beam is supported by the three suspender bars. Bars $AB$ and $EF$ are made of aluminum and bar $CD$ is made of steel. If each bar har a cross-sectional area of \qty{450}{\mm\squared}, determine the maximum value of $P$ if the allowable stress is $\left( \sigma_{\text{allow}} \right)_{\mathrm{st}} = \qty{200}{\MPa}$ for the steel and $\left( \sigma_{\text{allow}} \right)_{\mathrm{al}} = \qty{150}{\MPa}$ for the aluminum. $E_{\mathrm{st}} = \qty{200}{\GPa}, E_{\mathrm{al}} = \qty{70}{\GPa}$.
\end{problemwithimage}

\begin{derivation}
  \begin{figure} [ht]
    \centering
    \includegraphics[width=0.85\linewidth]{./figures/p12_1fbd.png}
    \caption{Free body diagram of the beam.}
  \end{figure}
  
  We start by considering the moment about $A$ giving us
  \begin{align*}
    \sum M_A &= 0 \\
    0 &= -P \cdot \qty{0,75}{\m} + F_{CD} \cdot \qty{1,5}{\m} - 2P \cdot \qty{2,25}{\m} + F_{EF} \cdot \qty{3}{\m}  \\
    \qty{5,25}{\m} \cdot P &= \qty{1,5}{\m} \cdot F_{CD} + \qty{3,0}{\m} \cdot F_{EF}
  .\end{align*}
  Using the moment equilibrium about $E$ yields,
  \begin{align*}
    \sum M_E &= 0 \\
    0 &= 2P \cdot \qty{0,75}{\m} - F_{CD} \cdot \qty{1,5}{\m} + P \cdot \qty{2,25}{\m} - F_{AB}\cdot \qty{3}{\m}  \\
    \qty{3,75}{\m} \cdot  P &= \qty{1,5}{\m} \cdot F_{CD} + \qty{3,0}{\m} \cdot F_{AB}
  .\end{align*}

  Using the fact that the bar is rigid the total system must deform as shown on \Cref{fig:p12_1def}.

  \begin{figure} [ht]
    \centering
    \includegraphics[width=0.65\linewidth]{./figures/p12_1def.png}
    \caption{Deformation of the system}
    \label{fig:p12_1def}
  \end{figure}

  From similar triangles we have that:
  \begin{align*}
    \frac{\delta_{CD} - \delta_{AB}}{\num{1,5}} &= \frac{\delta_{EF} - \delta_{AB}}{3} \\
    3 \delta_{CD} - 3 \delta_{AB} &= \num{1,5} \delta_{EF} - \num{1,5} \delta_{AB} \\
    3 \delta_{CD} &= \num{1,5} \delta_{EF} + \num{1,5} \delta_{AB} \\
    2 \delta_{CD} &= \delta_{EF} + \delta_{AB}
  .\end{align*}
  Now we can use the formula for deformation
  \[ 
  \delta = \frac{PL}{AE}
  \]
  and get
  \begin{align*}
    \frac{2 F_{CD} \cdot L}{A \cdot E_{\mathrm{st}}} &= \frac{L}{A \cdot E_{\mathrm{al}}} \left( F_{EF} + F_{AB} \right) \\
    F_{CD} &= \frac{E_{\mathrm{st}}}{2 \cdot E_{\mathrm{al}}} \left( F_{EF} + F_{AB} \right) \\
           &= \frac{\qty{200}{\GPa}}{2 \cdot \qty{70}{\GPa}} \left( F_{EF} + F_{AB} \right) \\
           &= \frac{10}{7} \left( F_{EF} + F_{AB} \right)
  .\end{align*}
  We now have three equations with three unknowns:
  \begin{align*}
    \num{1,5} \cdot F_{CD} + 3 \cdot F_{EF} &= \num{5,25} P \\
    \num{1,5} \cdot F_{CD} + 3 \cdot F_{AB} &= \num{3,75} P \\
    F_{CD} &= \frac{10}{7} \left( F_{EF} + F_{AB} \right)
  .\end{align*}
  Solving these gives:
  \begin{align*}
    F_{AB} &= \num{0,3676} P \\
    F_{CD} &= \num{1,7646} P \\
    F_{EF} &= \num{0,8676} P
  .\end{align*}
  Now the maximally allowable value of $P$ for each beam can be found separately using the following method
  \begin{align*}
    \left( \sigma_{\text{allow}} \right)_{\mathrm{al}} &= \frac{F_{AB}}{A} \\
    F_{AB} &= \left( \sigma_{\text{allow}} \right)_{\mathrm{al}} \cdot A \\
    \num{0,368} \left( P_{\text{allow}} \right)_{AB} &= \qty{150}{\MPa} \cdot \qty{450e-6}{\m\squared}  \\
    \left( P_{\text{allow}} \right)_{\mathrm{AB}} &= \frac{\qty{150}{\MPa} \cdot \qty{450e-6}{\m\squared} }{\num{0,368}} \\
    &= \qty{183,6}{\kN} 
  .\end{align*}
  Doing the same for the other suspenders yields:
  \begin{align*}
    \left( P_{\text{allow}} \right)_{\mathrm{CD}} &= \qty{77,8}{\kN} \\
    \left( P_{\text{allow}} \right)_{\mathrm{EF}} &= \qty{51,0}{\kN}
  .\end{align*}
  And from the above results it becomes obvious that suspender $EF$ will limit the allowable load to $P_{\mathrm{max}} = \left( P_{\text{allow}} \right)_{\mathrm{EF}} = \qty{51,0}{\kN}$.
\end{derivation}

\finalresult{P_{\mathrm{max}} = \qty{51,0}{\kN}}
