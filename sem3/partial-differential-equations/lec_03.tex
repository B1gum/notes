\begin{problem}{3.1}
Consider the function $f(x) = \cos x, 0<x<\pi$.

\begin{enumerate}[label=\alph*)]
  \item Sketch the expansion of $f$ and the corresponding Fourier series.
  \item Sketch the even expansion of $f$ and find the corresponding Fourier cosine series.
  \item Sketch the odd expansion of $f$ and find the corresponding Fourier cosine series.
  \item Calculate the error $E_N$ of the expansion and the odd expansion for $N = 1,2,3.$
    What can be said about the quality of the approximations for these two expansions=
\end{enumerate}
Hint: In some cases the expansion can be an even/odd function.

Hint: For $m, n \in \mathbb{N}, m \neq n, L > 0$:
\begin{align*}
  \int_{0}^{\frac{\pi}{2}} \cos x \sin(2nx) \, \mathrm{d}x &= - \frac{2n}{1 - 4 n^2} \\
  \int_{0}^{L} \cos \left( \frac{m \pi x}{L} \right) \sin \left( \frac{n \pi x}{L} \right) \, \mathrm{d}x  &= \frac{nL}{\pi \left( m^2 - n^2 \right)} \left( \cos \left( n \pi  \right) \cos \left( m \pi \right) - 1 \right) \\
  \int_{0}^{L} \cos \left( \frac{\pi x}{L} \right) \sin \left( \frac{\pi x}{L} \right) \, \mathrm{d}x &= 0
.\end{align*}
\end{problem}
