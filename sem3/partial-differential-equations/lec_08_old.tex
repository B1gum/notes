\lecture{8}{31. Oktober 2025}{Vibrating rectangular membranes and Double Fourier Sine Series}

\begin{problem}{8.1}
  Consider the function
  \[ 
  f(x,y) = 3 \sin \left( \frac{4 \pi y}{3} \right) \sin \left( \frac{3 \pi x}{2} \right) + 2 \sin \left( \frac{4 \pi y}{3} \right) \sin \left( \frac{3 \pi x}{6} \right), \qquad 0 \leq x \leq 2, 0 \leq y \leq 3
  .\]
  Find the Fourier coefficients of the double Fourier sine series of $f(x,y)$.

  \textit{Hint:} Coefficients of double Fourier sine series are unique.
\end{problem}

\begin{derivation}
  We start by simplifying the function as:
  \begin{align*}
    f(x,y) &= 3 \sin \left( \frac{4 \pi y}{3} \right) \sin \left( \frac{3 \pi x}{2} \right) + 2 \sin \left( \frac{4 \pi y}{3} \right) \sin \left( \frac{3 \pi x}{6} \right) \\
           &= 3 \sin \left( \frac{4 \pi y}{3} \right) \sin \left( \frac{3 \pi x}{2} \right) + 2 \sin \left( \frac{4 \pi y}{3} \right) \sin \left( \frac{\pi x}{2} \right)
  .\end{align*}
  Now we compare this to the general form of the Fourier sine series:
  \[ 
  f(x,y) = \sum_{m = 1}^{\infty} \sum_{n = 1}^{\infty} S_{m,n} \sin \left( \frac{n x y}{b} \right) \sin \left( \frac{m \pi x}{a} \right)
  .\]
  And see that:
\end{derivation}
\finalresult{
\begin{aligned}
  S_{3,4} &= 3 \\
  S_{1,4} &= 2 \\
  S_{m,n} &= 0 \quad \forall \quad (m,n) \notin \left\{ (3,4), (1,4) \right\}
.\end{aligned}
}

\begin{problem}{8.2}
  Consider the deflection $u(x,t)$ of a vibrating elastic membrane in the $xy$-plane, $0 \leq x \leq 1, 0 \leq y \leq 1$, fixed along its boundaries and governed by
  \[ 
  u_{t t}(x,y,t) = c^2 \left( u_{x x}(x,y,t) + u_{yy}(x,y,t) \right)
  .\]

  (a) Find the solution that corresponds to the initial conditions
  \begin{align*}
    u(x,y,0) &= \sin \left( 6 \pi x \right) \sin \left( 2 \pi y \right) \\
    u_t(x,y,0) &= 0
  .\end{align*}

  (b) Find the solution that corresponds to the initial conditions
  \begin{align*}
    u(x,y,0) &= 0 \\
    u_t(x,y,0) &= \sin \left( 6 \pi x \right) \sin \left( 2 \pi y \right)
  .\end{align*}

  (c) Find the solution that corresponds to the initial conditions
  \begin{align*}
    u(x,y,0) &= \sin \left( 6 \pi x \right) \sin \left( 2 \pi y \right) \\
    u_t(x,y,0) &= \sin \left( 6 \pi x \right) \sin \left( 2 \pi y \right)
  .\end{align*}

  \textit{Hint}: Coefficients of double Fourier dine series are unique.
\end{problem}

\begin{derivation}
  The general solution is given in the lecture as:
  \[ 
  u(x,y,t) = \sum_{m = 1}^{\infty} \sum_{n = 1}^{\infty} \left( E_{m,n} \cos \left( \lambda_{m,n} t \right) + J_{m,n} \sin \left( \lambda_{m,n}t \right)\right) \sin \left( \frac{m \pi x}{a} \right) \sin \left( \frac{n \pi y}{b} \right)
  .\]
  From the initial elongation we get:
  \begin{align*}
    u(x,y,0) &= \sin (6 \pi x) \sin (2 \pi y) \\
    \sin (6\pi x) \sin (2 \pi y) &= \sum_{m = 1}^{\infty} \sum_{n = 1}^{\infty}  E_{m,n} \sin \left( \frac{m \pi x}{a} \right) \sin \left( \frac{n \pi y}{b} \right) \\
    \implies \quad E_{6,2} &= 1
  .\end{align*}
  From the initial velocity we get:
  \begin{align*}
    u_t(x,y,0) &= 0 \\
    0 &= \sum_{m = 1}^{\infty} \sum_{n = 1}^{\infty} \lambda_{m,n}J_{m,n} \sin \left( \frac{m \pi x}{a} \right) \sin \left( \frac{n \pi y}{b} \right) \\
    \implies \quad J_{m,n} &= 0 \quad  \forall \quad(m,n)
  .\end{align*}
  To find the unique solution, we calculate:
  \begin{align*}
    \lambda_{m,n} &= c\pi \sqrt{\frac{m^2}{a^2} + \frac{n^2}{b^2}} \\
    \lambda_{6,2}&= c \pi \sqrt{6^2 + 2^2} \\
                 &= \sqrt{40}c \pi
  .\end{align*}
  And therefore, the unique solution becomes:
  \begin{align*}
    u(x,y,t) &= \sum_{m = 1}^{\infty} \sum_{n = 1}^{\infty}  \left( E_{m,n} \cos \left( \lambda_{m,n}t \right) + J_{m,n} \sin \left( \lambda_{m,n}t \right) \right) \sin \left( m\pi x \right) \sin \left( n \pi y \right) \\
             &= E_{6,2} \cos \left( \lambda_{6,2}t \right) \sin \left( 6 \pi x \right) \sin \left( 2\pi y \right) \\
             &= \cos \left( \sqrt{40} c\pi t \right) \sin \left( 6 \pi x \right) \sin \left( 2 \pi y \right)
  .\end{align*}


  If we use the elongation from (b) we get:
  \begin{align*}
    u(x,y,0) &= 0 \\
    0 &= \sum_{m = 1}^{\infty} \sum_{n = 1}^{\infty} E_{m,n} \sin \left( \frac{m \pi x}{a} \right) \sin \left( \frac{n \pi y}{b} \right) \\
    \implies E_{m,n} &= 0 \quad \forall \quad (m,n)
  .\end{align*}
  Using the initial velocity here, we obtain:
  \begin{align*}
    u_t(x,y,0) &= \sin \left( 6 \pi x \right) \sin \left( 2 \pi y \right) \\
    \sin (6 \pi x) \sin \left( 2 \pi y \right) &= \sum_{m = 1}^{\infty} \sum_{n = 1}^{\infty} \lambda_{m,n} J_{m,n} \sin \left( m\pi x \right) \sin (n \pi y) \\
    \implies \quad J_{6,2} \lambda_{6,2} &= 1 \\
    \quad J_{6,2} &= \frac{1}{\sqrt{40}c\pi}
  .\end{align*}
  The unique solution therefore becomes
  \[ 
  u(x,y,t) = \frac{1}{\sqrt{40}c \pi} \sin \left( \sqrt{40}c\pi t \right) \sin \left( 6 \pi x \right) \sin \left( 2 \pi y \right)
  .\]
  
  In case (c) we have found the corresponding coefficients of $E_{6,2} = 1$ and $J_{6,2} = 1 / (\sqrt{40}c \pi)$ in (a) and (b).
  The unique solution here will therefore be:
  \[ 
  u(x,y,t) = \left( \cos \left( \sqrt{40}c\pi t \right) + \frac{1}{\sqrt{40}c\pi} \sin \left( \sqrt{40}c\pi t \right) \right) \sin \left( 6 \pi x \right) \sin \left( 2 \pi y \right)
  .\]
\end{derivation}

\finalresult{
\begin{aligned}
  &(\mathrm{a}) & u(x,y,t) &= \cos \left( \sqrt{40}c \pi t \right) \sin \left( 6 \pi x \right) \sin \left( 2 \pi y \right) \\
  &(\mathrm{b}) & u(x,y,t) &= \frac{1}{\sqrt{40}c\pi} \sin \left( \sqrt{40}c\pi t \right) \sin \left( 6 \pi x \right) \sin \left( 2 \pi y \right) \\
  &(\mathrm{c}) & u(x,y,t) &= \left( \cos \left( \sqrt{40}c\pi t \right) + \frac{1}{\sqrt{40} c  \pi} \sin \left( \sqrt{40}c\pi t \right) \right) \sin \left( 6 \pi x \right) \sin \left( 2 \pi y \right)
.\end{aligned}
}

\begin{problem}{8.3}
  Consider the two-dimensional heat equation
  \[ 
  u_t(x,y,t) = c^2 (u_{x x}(x,y,t) + u_{yy}(x,y,t))
  .\]
  
  (a) Assume $u(x,y,t) = F(x,y)G(t)$ and derive a PDE for $F(x,y)$ and an ODE for $G(t)$.

  (b) Assume $F(x,y) = H(x) Q(y)$ to finally derive three ODEs for $H(x)$, $Q(y)$ and $G(t)$.
\end{problem}

\begin{derivation}
  We assume $u(x,y,t) = F(x,y) G(t)$ and insert to get:
  \begin{align*}
    F(x,y) G'(t) &= c^2 \left( F_{x x} (x,y) G(t) + F_{yy} (x,y) G(t) \right) \\
    \frac{G'(t)}{c^2 G(t)} &= \frac{F_{x x}(x,y) + F_{yy}(x,y)}{F(x,y)} = c_0 \\
    \implies G'(t) - c^2 c_0 G(t) &= 0 \\
    F_{x x}(x,y) + F_{yy}(x,y) - F(x,y) c_0 &= 0
  .\end{align*}
  
  Now we assume $F(x,y) = H(x)Q(y)$ to get
  \begin{align*}
    H''(x) Q(y) + H(x) Q''(y) - H(x)Q(y) c_0 &= 0 \\
    \frac{H''(x)}{H(x)} = - \frac{Q''(y)}{Q(y)} + c_0 &= c_1 \\
    \implies \quad H''(x) - c_1 H(x) &= 0 \\
    Q''(y) - Q(y)(c_0 - c_1) &= 0 \\
    G'(t) - c^2 c_0 G(t) &= 0
  \end{align*}
\end{derivation}

\finalresult{
  \begin{array}{cr}
  (\mathrm{a}) & \begin{aligned}
    F_{x x}(x,y) + F_{yy}(x,y) - F(x,y) c_0 &= 0 \\
    G'(t) - c^2 c_0 G(t) &= 0 \\[1em]
  \end{aligned} \\
  (\mathrm{b}) & \begin{aligned}
    H''(x) - c_1 H(x) &= 0 \\
    Q''(y) - Q(y) \left( c_0 - c_1 \right) &= 0 \\
    G'(t) - c^2 c_0 G(t) &= 0
  \end{aligned}
\end{array}
}


\begin{problem}{8.4}
  Consider the PDE
  \[ 
  u_t(x,y,z,t) = u_x(x,y,z,t) + u_y(x,y,z,t) + u_z(x,y,z,t)
  .\]
  Assume $u(x,y,z,t) = H(x)Q(y)P(z)G(t)$ and derive 4 ODEs for $H(x), Q(y), P(z),$ and $G(t)$.
\end{problem}

\begin{derivation}
  We start with the assumption:
  \[ 
  u(x,y,z,t) = A(x,y,z) G(t)
  .\]
  By insertion we obtain our first PDE as
  \begin{align*}
    G'(t) A(x,y,z) &= \left( A_x \left( x,y,z \right) + A_y \left( x,y,z \right) + A_z \left( x,y,z \right) \right) G(t) \\
    \frac{G'(t)}{G(t)} &= \frac{A_x(x,y,z) + A_y(x,y,z) + A_z(x,y,z)}{A(x,y,z)} = c_0 \\
    \implies \quad G'(t) - G(t)c_0 &= 0 \\
    A_x(x,y,z) + A_y(x,y,z) + A_z(x,y,z) - A(x,y,z) c_0 &= 0
  .\end{align*}
  
  Now we assume
  \[ 
  A(x,y,z) = B(x,y)P(z)
  .\]
  And obtain:
  \begin{align*}
      B_x(x,y) + B_y(x,y) + P'(z) - B(x,y)P(z) c_0 &= 0 \\
      \frac{B_x(x,y) + B_y(x,y)}{B(x,y)} = c_0 \frac{P'(z)}{P(z)} &= c_1\\
      \implies P'(z) - \left( c_0 - c_1 \right)P(z) &= 0 \\
       B_x(x,y) + B_y(x,y) - c_1 B(x,y) &= 0
  .\end{align*}
  
  Lastly, we assume $B(x,y) = H(x) Q(y)$ and obtain
  \begin{align*}
    H'(x) Q(y) + H(x) Q'(y) - c_1 H(x) Q(y) &= 0 \\
    \frac{H'(x)}{H(x)} = c_1 - \frac{Q'(y)}{Q(y)} &= c_2 \\
    \implies H'(x) - H(x) c_2 &= 0 \\
    Q'(y) - \left( c_1 - c_2 \right)Q(y) &= 0
  .\end{align*}
\end{derivation}

\finalresult{
\begin{aligned}
  &(\mathrm{a}) & H'(x) - c_2 H(x) &= 0 \\
  &(\mathrm{b}) & Q'(y) - (c_1 - c_2) Q_y &= 0 \\
  &(\mathrm{c}) & P'(z) - (c_0 - c_1) P(z) &= 0 \\
  &(\mathrm{d}) & G'(t) - G(t) c_0 &= 0
.\end{aligned}
}
