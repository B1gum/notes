\begin{problem} \label{prob:1}
%TC:ignore
Water flows steadily through a fire hose and nozzle.
The hose is \qty{35}{mm} in diameter and the nozzle tip is \qty{25}{mm} in diameter; water gauge pressure in the hose is \qty{510}{kPa}, and the stream leaving the nozzle is uniform.
The exit speed and pressure are \qty{32}{m/s} and atmospheric, respectively.

Find the force transmitted by the coupling between the nozzle and hose.
Indicate whether the coupling is in tension or compression.
State the assumptions made and sketch the situation.
%TC:endignore
\end{problem}
\begin{figure}[ht]
    \centering
    \incfig{p1_1}
    \caption{Sketch of the coupling with water flow direction and sign convention defined in  \Cref{eq:sgncon}.}
    \label{fig:p1_1}
\end{figure}

\begin{assumptions}
  \item Rigid and stationary coupling
  \item Steady flow
  \item Uniform flow at interfaces between coupling, hose, and nozzle
  \item Incompressible liquid
  \item Negligible wall shear and fluid weight in the short coupling relative to pressure and momentum
\end{assumptions}

\begin{derivation}
The momentum equation for an inertial control volume (as is the case for the coupling) is
\begin{equation} \label{eq:mominvec}
\Vec{F} = \Vec{F}_B + \Vec{F}_S = \frac{\partial }{\partial t} \int_{\mathrm{CV}} \Vec{V} \rho \, \mathrm{d}V + \int_{\mathrm{CS}} \Vec{V}\rho \Vec{V} \cdot \mathrm{d}\Vec{A}  
\end{equation}
where $\Vec{A}$ is the product of an area $A$ and an outward normal unit vector $\hat{n}$,
\begin{equation}\label{eq:sgncon}
  \Vec{A} = A \, \hat{n}
\end{equation}
as shown on \Cref{fig:p1_1}.In this case the flow is steady, hence $\frac{\partial }{\partial t} = 0$, furthermore, as we neglect body forces $\Vec{F}_B = 0$ on \Cref{eq:mominvec}.
Reducing the vector equation to a scalar equation in the $x$-direction (defined on \Cref{fig:p1_1}) we obtain
\[ 
F_x = F_{S_x} = \int_{CS} u \rho \Vec{V} \cdot \mathrm{d}\Vec{A}
.\]
As the flow is assumed uniform at both the inlet and outlet the integral further reduces to a sum as
\begin{equation}\label{eq:momste1d}
  F_x = F_{S_x} = \sum_{\mathrm{CS}} u \rho \Vec{V} \cdot \Vec{A}
\end{equation}
The forces in the $x$-direction on the control volume are the pressure at the interface between the hose and the coupling ($+p_h A_h$), the pressure at the nozzle ($-p_n A_n$), and the reaction on the coupling $R_x$.
Here the signs of these terms is determined based on the sign convention defined in \Cref{eq:sgncon}.
From \Cref{eq:momste1d} we get
\begin{equation} \label{eq:momste1dsub}
  R_x + p_h A_h - p_n A_n = \rho Q \left( V_n - V_h \right)
\end{equation}
As we are using gauge pressures $p_n = \qty{0}{\kilo\pascal} \implies p_n A_n = \qty{0}{\newton}$, therefore \Cref{eq:momste1dsub} reduces to
\begin{equation} \label{eq:momste1dsubred}
  R_x + p_h A_h = \rho Q \left( V_n - V_h \right)
\end{equation}
The continuity equation prescribes that the flow rate $Q = V A = \mathrm{const.}$
Hence,
\begin{subequations} \label[pluralequation]{eq:cont}
  \begin{align}
  Q &= A_n V_n \label{eq:conta}\\
  V_h &= \frac{Q}{A_h} = \frac{A_n V_n}{A_h}
\end{align}
\end{subequations}
Substituting \Cref{eq:cont} into \Cref{eq:momste1dsubred} and solving for $R_x$ we obtain
\begin{align*}
  R_x + p_h A_h &= \rho A_n V_n \left( V_n - \frac{A_n V_n}{A_h} \right) \\
  R_x &= \rho A_n V_n^2 \left( 1 - \frac{A_n}{A_h} \right) - p_h A_h \\
  R_x &= \qty{1000}{\kilogram\per\cubic\meter} \cdot \left( \pi \cdot \left( \qty{0,0125}{\meter}  \right)^2 \right) \cdot \left( \qty{32}{\meter\per\second}  \right)^2 \left( 1 - \frac{ \left( \qty{0,0125}{\meter}  \right)^2 }{\left( \qty{0,0175}{\meter}  \right)^2} \right) - \qty{510}{\kilo\pascal} \cdot \pi \cdot \left( \qty{0,0175}{\meter} \right)^2 \\
  R_x &= \qty{-244,48}{\newton} 
.\end{align*}
The negative sign (with $+x$ being to the right as shown on \Cref{fig:p1_1}) means the coupling reaction force points to the left, and the joint will therefore be in tension.
\end{derivation}

\finalresult{R_c = \qty{245}{\newton} \, \left( \text{Tension} \right)}


\begin{problemwithimage}[0.35\linewidth]{./figures/p2_1.png}
%TC:ignore
  Determine the gauge pressure in \unit{\kilo\pascal} at point $a$, if liquid $A$ has $\mathrm{SG} = \num{1,20} $ and liquid $B$ has $\mathrm{SG} = \num{0,75}$.
  The liquid surrounding point $a$ is water, and the tank on the left is open to the atmosphere.
%TC:endignore
\end{problemwithimage}

\begin{assumptions}
  \item Static equilibrium
  \item All fluids are at rest and will not mix and will not exert any capillary effects on each other
  \item All liquids have constant densities 
  \item The left tank is open to the atmosphere, i.e. $p_b = 0$ gauge at the surface
\end{assumptions}

\begin{figure}[ht]
    \centering
    \incfig[0.9]{p2_2}
    \caption{Sketch of the static tank and pipe system.}
\end{figure}

\begin{derivation}
For an incompressible fluid with density $\rho$ the pressure $p$ scales linearly with the depth $h$ as
\[ 
\Delta p = \rho g h
.\]
Given two points separated by a series of fluids with densities and heights of $\rho_i$ and $h_i$, respectively, the total change in pressure $\Delta p$ is simply
\begin{equation}\label{eq:delp}
  \Delta p = g \sum_i \rho_i h_i
\end{equation}
Starting at point $a$ we go a distance $h_1$ ``up'' in the water corresponding to a height of
\[ 
h_1 = \qty{0,25}{\meter} (\text{up to the }A \text{ /water-interface}) + \qty{0,125}{\meter} = \qty{0,375}{\meter} 
.\]
This corresponds to a pressure difference of
\[ 
p_1 = p_a - \rho_w g h_1
.\]
Inside liquid $A$ a distance of $h_2 = \qty{0,25}{\meter}$ must be traversed vertically down to reach the $A$/$B$-interface.
This gives a pressure difference of
\[ 
p_2 = p_1 + \rho_{A} g h_2
.\]
Now in liquid $B$ the net rise to the open tank surface is $h_3 = \qty{0,9}{\meter} - \qty{0,4}{\meter} = \qty{0,5}{\meter}$. Which gives our atmospheric (gauge) pressure
\[ 
p_{\mathrm{atm}} = p_b = p_2 - \rho_B g h_3 = 0
.\]
Combining these per \Cref{eq:delp} yields
\begin{align*}
  p_a - \rho_w g h_1 + \rho_{A} g h_2 - \rho B g h_3 &= 0 \\
  \implies p_a &= \rho_w g \left( h_1 - \mathrm{SG}_A h_2 + \mathrm{SG}_B h_3 \right) \\
    &= \qty{1000}{\kilogram\per\cubic\meter} \cdot \qty{9,81}{\meter\per\square\second} \left( \qty{0,375}{\meter} - \num{1,2} \cdot \qty{0,25}{\meter} + \num{0,75} \cdot \qty{0,5}{\meter} \right) \\
    &= \qty{4,41}{\kilo\pascal}
.\end{align*}
The gauge pressure at $a$ is therefore $p_a = \qty{4,41}{\kilo\pascal}$.
\end{derivation}

\finalresult{p_a = \qty{4,41}{\kilo\pascal} \, (\text{gauge})}


\begin{problem}
%TC:ignore
 Determine the stream function $\psi$ that will yield the velocity field.
\[ 
\Vec{V} = 2y \left( 2x + 1 \right) \hat{i} + \left( x \left( x + 1 \right) - 2y^2 \right) \hat{j} \, V
.\] 
%TC:endignore
\end{problem}

\begin{assumptions}
 \item Incompressible flow in the $xy$-plane
\end{assumptions}

\begin{derivation}
The stream function for two-dimensional incompressible flow is defined as
\begin{equation}\label{eq:strfun}
  u \equiv \frac{\partial \psi}{\partial y} \qquad \text{and} \qquad v \equiv - \frac{\partial \psi}{\partial x}
\end{equation}
Starting with the $u$-coordinate we can integrate with respect to $y$ as
\begin{align*}
  \frac{\partial \psi}{\partial y} &= 2y \left( 2x + 1 \right) \\
  \psi(x,y) &= \int 2y (2x + 1) \, \mathrm{d}y \\
            &= 2xy^2 + y^2 + f(x)
.\end{align*}
Doing the same with the $v$-coordinate but integrating with respect to $x$ we obtain
\begin{align*}
  -\frac{\partial \psi}{\partial x} &= x \cdot \left( x+1 \right) - 2y^2 \\
  \psi(x,y) &= -\int \left( x \cdot (x+1)- 2y^2 \right) \, \mathrm{d}x  \\
            &= - \frac{x^3}{3} - \frac{x^2}{2} + 2xy^2 + g(y)
.\end{align*}
Comparing these we see that $f(x) = - \frac{x^3}{3} - \frac{x^2}{2}$ and $g(y) = y^2$.
Hence the stream function is
\[ 
\psi(x,y) = y^2 + 2xy^2 - \frac{x^2}{2} - \frac{x^3}{3}
.\]
This solution can be verified by checking that the found stream function does not break the definition of the stream function presented in \Cref{eq:strfun} as
\begin{align*}
  u(x,y) &= \frac{\partial }{\partial y} \left( y^2 + 2xy^2 - \frac{x^3}{3} - \frac{x^2}{2} \right) & &\implies & u(x,y) &= 2y + 4xy = 2y(2x + 1) \\
  v(x,y) &= \frac{\partial}{\partial x} \left( y^2 + 2xy^2 - \frac{x^3}{3} - \frac{x^2}{2} \right) & &\implies & v(x,y) &= x^2 + x - 2y^2 = x(x+1) - 2y^2
.\end{align*}
Which means the stream function is correct.
\end{derivation}

\finalresult{\psi(x,y) = y^2 + 2xy^2 - \frac{x^3}{3} - \frac{x^2}{2}}


\begin{problemwithimage}[0.5\linewidth]{./figures/p4_1.png}
%TC:ignore
Water flows steadily through the nozzle, shown in the figure, discharging to atmosphere.
Calculate the horizontal component of force in the flanged joint.
Indicate whether the joint is in tension or compression.
Use density of $\mathrm{water} = \qty{1000}{\kilogram\per\cubic\meter}$.
%TC:endignore
\end{problemwithimage}

\begin{assumptions}
  \item Steady flow
  \item Incompressible flow
  \item Uniform flow
  \item Negligible wall shear and body forces in the (relatively) small joint
  \item The joint discharges into the atmosphere; $p_2 = 0$
  \end{assumptions}

\begin{figure}[ht]
    \centering
    \incfig{p4_2}
    \caption{Sketch of the flanged joint with flow- and area normal directions indicated.}
    \label{fig:p4_2}
\end{figure}

\begin{derivation}
We can once again apply the momentum equation for an inertial control volume, \Cref{eq:mominvec},
\[ 
\Vec{F} = \Vec{F}_B + \Vec{F}_S = \frac{\partial }{\partial t} \int_{\mathrm{CV}} \Vec{V} \rho \, \mathrm{d}V + \int_{\mathrm{CS}} \Vec{V} \rho \Vec{V} \cdot \mathrm{d}\Vec{A}
.\]
Following the same argumentation as in \Cref{prob:1} with the same assumptions this reduces to the sum from \Cref{eq:momste1d}
\[ 
F_x = F_{S_x} = \sum_{\mathrm{CS}} u \rho \Vec{V} \cdot \Vec{A}
.\]
Hence,
\begin{subequations} \label[pluralequation]{eq:p4mom}
  \begin{align}
    R_x + p_1 \cdot A_1 + p_2 \cdot A_2 &= V_1 \cdot \left( -\rho \cdot V_1 \cdot A_1 \right) + V_2 \cdot \cos \theta \cdot \left( \rho \cdot V_2 \cdot A_2 \right) \\
    R_x &= - p_1 \cdot A_1 + \rho \left( V_2^2 \cdot \cos \theta \cdot A_2 - V_1^2 A_1 \right)  \label{eq:momp4}
  .\end{align}
\end{subequations}
In \Cref{eq:p4mom} it is used that $p_2 = 0 \, (\text{gauge}) \implies p_2 \cdot A_2 = 0$. From the continuity equation we have that
\begin{align*}
  A_2 V_2 &= A_1 V_1 \\
  V_2 &= V_1 \frac{A_1}{A_2} \\
      &= V_1 \left( \frac{D_1}{D_2} \right)^2 \\
      &= \qty{1,5}{\meter\per\second} \cdot \left( \frac{\qty{0,30}{\meter} }{\qty{0,15}{\meter}} \right)^2 \\
      &= \qty{6}{\meter\per\second} 
.\end{align*}
Now plugging all the known values into \Cref{eq:momp4} we obtain
\begin{align*}
  R_x &= \qty{-15}{\kilo\pascal} \cdot \pi \cdot \frac{\left( \qty{0,30}{\meter}  \right)^2}{4} + \qty{1000}{\kilogram\per\cubic\meter} \cdot \left( \left( \qty{6}{\meter\per\second}  \right)^2 \cdot \cos \ang{30} \cdot \pi \cdot \frac{\left( \qty{0,15}{\meter}  \right)^2}{4} - \left( \qty{1,5}{\meter\per\second}  \right)^2 \cdot \pi \cdot \frac{\left( \qty{0,30}{\meter}  \right)^2}{4}\right) \\
      &= \qty{-668,39}{\newton} 
.\end{align*}
Therefore, the reaction force in the joint will be in the $-x$-direction, as defined on \Cref{fig:p4_2}, and the joint will be in tension.
This aligns with the expectation given that it needs to withstand high pressure and a large change in momentum of the fluid.  
\end{derivation}

\finalresult{R_x = \qty{668}{\newton} \,  (\text{Tension})}




\begin{problem}
%TC:ignore
The velocity components for an incompressible steady flow field are given below
\[ 
u = A \left( x^2 + z^2 \right) \qquad v = B \left( xy + yz \right)
.\]
Determine the $z$-component of velocity for steady and unsteady flow.
%TC:endignore
\end{problem}

\begin{assumptions}
  \item Incompressible flow
\end{assumptions}

\begin{derivation}
The continuity equation on differential form can be written as
\begin{equation} \label{eq:convec}
  \nabla \cdot \rho \Vec{V} + \frac{\partial \rho}{\partial t} = 0
\end{equation}
As we are assuming the flow to be incompressible $\rho = \mathrm{const.}$
This means that $\frac{\partial \rho}{\partial t} = 0$ and $\nabla \cdot \rho \Vec{V} = \nabla \cdot \Vec{V}$ in \Cref{eq:convec} which therefore reduces to
\begin{equation} \label{eq:convecinc}
  \nabla \cdot \Vec{V} = 0
\end{equation}
for an incompressible flow.
On Cartesian form \Cref{eq:convecinc} becomes
\[ 
\frac{\partial u}{\partial x} + \frac{\partial v}{\partial y} + \frac{\partial w}{\partial z} = 0
.\]
Substituting in our known values we get
\begin{align*}
\frac{\partial }{\partial x} \left( A \left( x^2 + z^2 \right) \right) + \frac{\partial}{\partial y} \left( B \left( xy + yz \right) \right) + \frac{\partial w}{\partial z} &= 0 \\
  - \frac{\partial w}{\partial z} &= 2Ax + B \left( x+z \right) \\
  \frac{\partial w}{\partial z} &= -2Ax - Bx - Bz \\
  w &= - \int \left( 2A + B \right)x \, \mathrm{d}z - \int Bz \, \mathrm{d}z \\
  &= - \left( 2A + B \right)xz - \frac{B}{2} z^2 + f(x,y)
\end{align*}
where $f(x,y)$ is a general function of $x$ and $y$. The value of this can only be determined given additional boundary conditions.
As the time-variant term $\frac{\partial \rho}{\partial t} = 0$ in \Cref{eq:convec} for both steady and unsteady flows, as long as they are incompressible as in this case $\rho = \mathrm{const.}$
Therefore the general expression for the $z$-component of the velocity will be the same for both steady and unsteady flow. The unknown function $f(x,y)$ could however also be a function of $t$ for the unsteady flow. Hence

\finalresult{
\begin{aligned}
  w &= - (2A + B) xz - \frac{B}{2}z^2 + f(x,y) & &(\text{steady}) \\
  w &= - (2A + B) xz - \frac{B}{2}z^2 + f(x,y,t) & &(\text{unsteady})
.\end{aligned}}


\end{derivation}


\begin{problemwithimage}[0.35\linewidth]{./figures/p6_1.png}
%TC:ignore
The water flow rate through the vertical bend shown in the figure below is $\qty{2,83}{\cubic\meter\per\second}$.
Calculate the magnitude, direction, and location of the resultant force of the water pipe bend.
Use density of $\mathrm{water}= \qty{999}{\kilogram\per\cubic\meter}$.
%TC:endignore
\end{problemwithimage}

\begin{assumptions}
\item Rigid and stationary bend
\item Incompressible, steady and uniform flow
\item Negligible wall shear in the (relatively) short bend
\end{assumptions}

\begin{figure}[ht]
\centering
    \incfig{p6_2}
    \caption{Sketch of the vertical bend with flow-, area normal- and external force directions.}
    \label{fig:p6_2}
\end{figure}

\begin{derivation}
The continuity equation is generally
\begin{equation}\label{eq:contgen}
  \frac{\partial}{\partial t} \int_{\mathrm{CV}} \rho \, \mathrm{d}V + \int_{\mathrm{CS}} \rho \Vec{V} \cdot \mathrm{d} \Vec{A} = 0
.\end{equation}
For a steady incompressible flow \Cref{eq:contgen} reduces to \Cref{eq:conta},
\[ 
Q = A_n V_n = \mathrm{const.}
.\]
For any point $n$.
Using this we can calculate the flow velocity through the bend, as
\[ 
V = \frac{Q}{A} = \frac{Q}{\pi\cdot \frac{d^2}{4}} = \frac{\qty{2,83}{\cubic\meter\per\second} }{\pi \cdot \frac{\left( \qty{0,9}{\meter}  \right)^2}{4}} = \qty{4,45}{\meter\per\second}
.\]
As shown in \Cref{eq:conta} flow speed $V$ will be constant for constant area $A = A_1 = A_2$, hence the flow speed through the entire bend must be $V = V_1 = V_2 = \qty{4,45}{\meter\per\second}$.

The Bernoulli equation is
\begin{equation} \label{eq:berneq}
  \frac{p}{\rho} + \frac{V^2}{2} + gz = \mathrm{constant}
.\end{equation}
Using the Bernoulli equation, \Cref{eq:berneq}, we can find the pressure $p_2$ at the outlet of the bend as
\begin{align*}
  \frac{p_1}{\rho} + \frac{V^2}{2} + g \cdot 0 &= \frac{p_2}{\rho} + \frac{V^2}{2} + gz \\
  \frac{p_2}{\rho} &= \frac{p_1}{\rho} - gz \\
  p_2 &= p_1 - \rho g z \\
      &= p_1 - \rho g \left( r + \frac{d}{2} \right) \\
      &= \qty{34,5}{\kilo\pascal} - \qty{999}{\kilogram\per\cubic\meter} \cdot \qty{9,81}{\meter\per\square\second} \cdot \left( \qty{0,6}{\meter} + \frac{\qty{0,9}{\meter} }{2} \right) \\
      &= \qty{24,2}{kPa} 
.\end{align*}
Note that in the above the height difference $z$ has been calculated as the centerline elevation difference $z = r + \frac{d}{2}$.
Now we can use the momentum equation for an inertial control volume, \Cref{eq:mominvec},
\[ 
\Vec{F} = \Vec{F}_B + \Vec{F}_S = \frac{\partial }{\partial t} \int_{CV} \Vec{V} \rho \, \mathrm{d}V + \int_{\mathrm{CS}} \Vec{V} \rho \Vec{V} \cdot \mathrm{d}\Vec{A}
.\]
In the case of steady flow \Cref{eq:mominvec} reduces to
\begin{equation}\label{eq:mominvecste}
    \Vec{F}_B + \Vec{F}_S = \int_{CS} \Vec{V} \rho \Vec{V} \cdot \mathrm{d} \Vec{A}
.\end{equation}
Splitting \Cref{eq:mominvecste} into Cartesian coordinates yields
\begin{subequations} \label[pluralequation]{eq:mominsvecstecar}
  \begin{align}
  \left( F_B \right)_x + \left( F_S \right)_x &= \int_{\mathrm{CS}} u \rho \Vec{V} \cdot \mathrm{d}\Vec{A} \label{eq:mominsvecstex} \\ 
  \left( F_B \right)_y + \left( F_S \right)_y &= \int_{\mathrm{CS}} v \rho \Vec{V} \cdot \mathrm{d}\Vec{A} \label{eq:mominsvecstey}
.\end{align}
\end{subequations}
As $V = V_1 = V_2 = \qty{4,45}{\meter\per\second}$, $u = v = V = \qty{4,45}{\meter\per\second}$ in \Cref{eq:mominsvecstecar}.
Using the convention from \Cref{eq:sgncon} we can express \Cref{eq:mominsvecstex} as
\begin{align*}
  R_x + \left( F_B \right)_x + \left( F_S \right)_x &= V^2 \cdot \rho \cdot A \\
  R_x - p_2 A &= V^2 \cdot \rho \cdot A \\
  R_x &= \pi \cdot \frac{d^2}{4} \left( V^2 \cdot \rho + p_2 \right) \\
      &= \pi \cdot \frac{\left( \qty{0,9}{\meter}  \right)^2}{4} \left( \left( \qty{4,45}{\meter\per\second}  \right)^2 \cdot \qty{999}{\kilogram\per\cubic\meter} + \qty{24,2}{\kilo\pascal} \right) \\
  &= \qty{28,0}{\kilo\newton}
.\end{align*}
I.e. the water will experience a reaction force of $R_x = \qty{28,0}{\kilo\newton}$ from the bend, which means that the water will exert a force $F_x = - R_x = \qty{-28,0}{\kilo\newton}$ on the bend.

From \Cref{eq:mominsvecstey} we get
\begin{align*}
  R_y + \left( F_B \right)_y + \left( F_S \right)_y &= - V^2 \cdot \rho \cdot A \\
  R_y - \left( \rho g A \frac{2\pi}{4} h \right) + \left( p_1 A \right) &= - V^2 \cdot \rho\cdot A \\
  R_y &= \rho g A \frac{2\pi}{4}h - V^2 \cdot \rho \cdot A - p_1 A \\
      &= A \left( \rho g \frac{\pi}{2}h - V^2 \cdot \rho - p_1 \right) \\
      &= \pi \cdot \frac{d^2}{4} \left( \rho \left( g \frac{\pi}{2} \left( r + \frac{d}{2} \right) - V^2 \right) - p_1 \right) \\
      &= \pi \cdot \frac{\left( \qty{0,9}{\meter} \right)^2}{4} \left( \qty{999}{\kg\per\cubic\m} \left( \qty{9,81}{\m\per\square\s} \cdot \frac{\pi}{2} \left( \qty{0,6}{\m} + \frac{\qty{0,9}{\m} }{2} \right) - \left( \qty{4,45}{\meter\per\s}  \right)^2  \right) - \qty{34,5}{\kilo\Pa}  \right) \\
      &= \qty{-24,2}{\kilo\newton}
.\end{align*}
This means the water will experience a reaction force of $R_y = \qty{-24,2}{\kilo\newton}$ from the bend, hence, the water will exert a force $F_y = - R_y = \qty{24,2}{\kilo\newton}$ on the bend.
In the above calculation the body force $\left( F_B \right)_y$ has been calculated as the gravitational force on the water in the bend.
Using $F_x$ and $F_y$ the total force exerted by the water on the bend can be computed as
\[ 
F = \sqrt{F_x^2 + F_y^2} = \sqrt{\left( -\qty{28,0}{\kilo\newton}  \right)^2 + \left( \qty{24,2}{\kilo\newton}  \right)^2} = \qty{37,0}{\kilo\newton} 
.\]
The direction of the force can be found using trigonometry as
\begin{align*}
  \tan \theta &= \frac{F_y}{F_x} \\
  \theta &= \tan^{-1} \left( \frac{F_y}{F_x} \right)\\
  &= \tan^{-1} \left( \frac{\qty{24,2}{\kilo\newton} }{\qty{-28}{\kilo\newton} } \right) \\
  &= \ang{139,16}
.\end{align*}
This angle is measured with respect to the $+x$-direction and the force will therefore be in the direction of the 2nd quadrant.
\end{derivation}

\finalresult{F = \qty{37,0}{\kilo\newton} \, \angle \,  \ang{139}}


\begin{problem}
%TC:ignore
A viscous liquid is sheared between two parallel disks of radius $R$, one of which rotates while the other is fixed.
The velocity field is purely tangential, and the velocity varies linearly with $z$ from $V_{\phi} = 0$ at $z = 0$ (the fixed disk) to the velocity of the rotating disk at its surface ($z = h$).

Derive an expression for the velocity field between the disks.
%TC:endignore
\end{problem}

\begin{assumptions}
\item Incompressible flow
\item Purely tangential flow
\item Velocity varies linearly with $z$
\item No slip
\end{assumptions}

\begin{figure}[ht]
    \centering
    \incfig[0.7]{p7_1}
    \caption{Sketch}
    \label{fig:p7_1}
\end{figure}

\begin{derivation}
Here we will apply the continuity equation on cylindrical form
\begin{equation}\label{eq:contcyn}
  \frac{1}{r} \frac{\partial}{\partial r} r \rho V_r + \frac{1}{r}\frac{\partial }{\partial \phi} \rho V_{\phi} + \frac{\partial }{\partial z} \rho V_z + \frac{\partial \rho}{\partial t} = 0
\end{equation}
As the flow is assumed incompressible $\frac{\partial \rho}{\partial t} = 0$ in \Cref{eq:contcyn} and the assumption of purely tangential flow means $V_r = V_z = 0$ in \Cref{eq:contcyn}.
Hence the continuity equation reduces to
\[ 
  \frac{\partial V_{\phi}}{\partial \phi} = 0
.\]
This means that the velocity $V_{\phi}$ is not a function of $\phi$ and hence $V_{\phi} = V_{\phi} \left( r, z \right)$.
As we know the velocity scales linearly with $z$ we can write the function for $z$ as a normal linear function,
\begin{equation} \label{eq:linvel}
 V_{\phi} \left( r, z \right) = z \cdot f(r) + C
\end{equation}
Applying the given boundary conditions of $V_{\phi} \left( r, 0 \right) = 0$ and $V_{\phi} \left( r, h \right) = r \omega$ to \Cref{eq:linvel} yields
  \begin{align}
  V_{\phi} \left( r, 0 \right) &= 0 \nonumber \\
  0 \cdot f(r) + C &= 0 \nonumber \\
  \implies C &= 0 \nonumber \\
  V_{\phi} \left( r, h \right) &= r \omega \nonumber \\
  \implies h \cdot f(r) &= r\omega \nonumber \\
  \implies f(r) &= \frac{r\omega}{h} \label{eq:fr}
.\end{align}
Substituting \Cref{eq:fr} into \Cref{eq:linvel} gives an expression for the tangential velocity $V_{\phi}$
\begin{equation}\label{eq:vphi}
    V_{\phi} = \frac{r\omega}{h}z
.\end{equation}
The velocity flow field on cylindrical coordinates is
\begin{equation}\label{eq:velcyn}
  \Vec{V} = V_r \,  \hat{e}_r + V_{\phi} \, \hat{e}_{\phi} + V_z \,  \hat{e}_z
\end{equation}
In \Cref{eq:velcyn} $V_r = V_z = 0$ since the flow is assumed purely tangential.
Therefore the entire velocity flow field is given by substituting \Cref{eq:vphi} into \Cref{eq:velcyn},
\[ 
\Vec{V} = V_{\phi} \, \hat{e}_{\phi} = \frac{r\omega}{h}z \, \hat{e}_{\phi}
.\]
\end{derivation}

\finalresult{\Vec{V}(r,z,\phi) = \frac{r\omega}{h}z \,  \hat{e}_{\phi}}
