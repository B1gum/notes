\lecture{8}{17. September}{Bernoulli Equation Continued}

\begin{problemwithimage}{p8_1}{6.23}
  Two water reservoirs at a \qty{30}{\m} elevation each have discharge pipes that are connected at a ``tee'' junction.
  One pipe has a \qty{200}{\mm} diameter and the other a \qty{150}{\mm} diameter.
  The outlet pipe from the ``tee'' is \qty{300}{\mm} in diameter and discharges to the atmosphere at an elevation of \qty{20}{\m}.
  Determine the total flow rate.
\end{problemwithimage}

\begin{assumptions}
  \item Inviscid, incompressible, steady, immiscible, and uniform flow
\end{assumptions}

\begin{derivation}
  We consider a streamline from reservoir 1 and to the discharge, which we call point 4. Under the assumptions of inviscid and immiscible flow the Bernoulli equation can be applied along this streamline
  \[ 
  \frac{p_1}{\rho} + \frac{V_1^2}{2} + gz_1 = \frac{p_4}{\rho} + \frac{V_4^2}{2} + gz_4
  .\]
  In this case $p_1 = V_1 = p_4 = 0$ and therefore the outlet velocity can be found as
  \begin{align*}
    gz_1 &= \frac{V_4^2}{2} + gz_4 \\
    V_4 &= \sqrt{2g \left( z_1 - z_4 \right)} \\
    &= \sqrt{2 \cdot \qty{9,81}{\m\per\square\s} \cdot \left( \qty{30}{\m} - \qty{20}{\m} \right)} \\
    &= \qty{14,0}{\m\per\s} 
  .\end{align*}
  Knowing the flow speed and the pipe diameter the flow rate can be found using the continuity equation as
  \[ 
  Q = A_4 V_4 = \pi \cdot \frac{\left( \qty{300}{\mm}  \right)^2}{4} \cdot \qty{14,0}{\m\per\s} = \qty{0,99}{\cubic\m\per\s} 
  .\]
\end{derivation}

\finalresult{Q = \qty{0,99}{\cubic\m\per\s} }

\begin{problemwithimage}{p8_2}{6.28}
  The turbine extracts power from the water flowing from the reservoir.
  Find the horsepower extracted if the flow through the system is \qty{1000}{ft^3\per\s}.
  Draw the energy line and the hydraulic grade line.
\end{problemwithimage}

\begin{assumptions}
  \item Steady, incompressible, inviscid and uniform flow
\end{assumptions}

\begin{derivation}
  The flow speed through the turbine can be found from the continuity equation
  \[ 
  Q = A_2 V_2 \implies V_2 = \frac{Q}{A_2} = \frac{\qty{1000}{\frac{ft^3}{s}} }{\pi \cdot \left( \qty{6}{in.}  \right)^2} = \qty{2,7}{\m\per\s} 
  .\]
  From the energy equation we get
  \[ 
  \frac{p_1}{\rho} + \frac{V_1^2}{2} + gz_1 = \frac{p_2}{\rho} + \frac{V_2^2}{2} + gz_2 + T
  .\]
  In this $p_1 = p_2 = V_1 \approx 0$, hence
  \[ 
  gz_1 = \frac{V_2^2}{2} + gz_2 + T \implies T = g \left( z_1 - z_2 \right) - \frac{V_2^2}{2}
  .\]
  Dividing this by $g$ yields the total head
  \[ 
  H = z_1 - z_2 - \frac{V_2^2}{2 g} = \qty{1000}{ft} - \qty{500}{ft}  - \frac{\left( \qty{2,7}{\m\per\s}  \right)^2}{2 \cdot \qty{9,81}{\m\per\square\s} } = \qty{152,03}{\m} 
  .\]
  The formula for finding the hydraulic power from a given head is
  \[ 
  P_{\mathrm{hyd}} = \rho g Q H = \qty{1000}{\kg\per\cubic\m}  \cdot \qty{9,81}{\m\per\s\squared}  \cdot \qty{1000}{\frac{ft^3}{\s}}  \cdot \qty{152,03}{\m} = \qty{42,232}{\kW}  
  .\]
\end{derivation}

\finalresult{P_\text{hyd} = \qty{42,2}{\kW}}
